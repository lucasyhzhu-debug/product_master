```typescript
import type { QueryCtx } from "../_generated/server";
import type { Doc, Id } from "../_generated/dataModel";

/**
 * Build a map of menuProductId -> total production pieces per unit of that product.
 *
 * Dynamic iteration over ALL componentTypes where category="production" AND unit="pcs".
 * NEVER hardcode BIG_BALL / MID_BALL here — future production types (Hazelnut, new
 * stuffed variants) must be counted automatically. This is the single source of
 * truth for the "units sold = BOM production pieces" rule used across analytics.
 */
export async function getProductionUnitsPerProduct(
  ctx: QueryCtx,
): Promise<Map<Id<"menuProducts">, number>> {
  const allComponentTypes = await ctx.db.query("componentTypes").collect();
  const productionTierOneIds = new Set<string>();
  for (const ct of allComponentTypes) {
    if (ct.category === "production" && ct.unit === "pcs") {
      productionTierOneIds.add(ct._id as string);
    }
  }

  const allBom = await ctx.db.query("menuProductComponents").collect();
  const result = new Map<Id<"menuProducts">, number>();
  for (const row of allBom) {
    if (!productionTierOneIds.has(row.componentTypeId as string)) continue;
    const mpId = row.menuProductId;
    result.set(mpId, (result.get(mpId) ?? 0) + row.quantity);
  }
  return result;
}

/**
 * Per-type breakdown: menuProductId -> { componentTypeCode -> pcs }.
 * Used by D1 (units stacked by type) and D4 (mix over time).
 */
export async function getProductionUnitsByTypePerProduct(
  ctx: QueryCtx,
): Promise<{
  byProduct: Map<Id<"menuProducts">, Map<string, number>>;
  typeCodes: string[]; // ordered, for stable legend
  typeCodeToName: Map<string, string>;
}> {
  const allComponentTypes = await ctx.db.query("componentTypes").collect();
  const productionTypes = allComponentTypes
    .filter((ct) => ct.category === "production" && ct.unit === "pcs")
    .sort((a, b) => (a.sortOrder ?? 999) - (b.sortOrder ?? 999));

  const idToCode = new Map<string, string>();
  const codeToName = new Map<string, string>();
  const typeCodes: string[] = [];
  for (const ct of productionTypes) {
    idToCode.set(ct._id as string, ct.code);
    codeToName.set(ct.code, ct.name);
    typeCodes.push(ct.code);
  }

  const allBom = await ctx.db.query("menuProductComponents").collect();
  const byProduct = new Map<Id<"menuProducts">, Map<string, number>>();
  for (const row of allBom) {
    const code = idToCode.get(row.componentTypeId as string);
    if (!code) continue;
    const mpId = row.menuProductId;
    if (!byProduct.has(mpId)) byProduct.set(mpId, new Map());
    const m = byProduct.get(mpId)!;
    m.set(code, (m.get(code) ?? 0) + row.quantity);
  }

  return { byProduct, typeCodes, typeCodeToName: codeToName };
}

/**
 * Count total production units for a given order item, resolving via BOM.
 * Returns 0 if the orderItem has no linked menuProduct (manual items don't count).
 */
export function unitsForOrderItem(
  item: Pick<Doc<"orderItems">, "menuProductId" | "quantity">,
  unitsPerProduct: Map<Id<"menuProducts">, number>,
): number {
  if (!item.menuProductId) return 0;
  const per = unitsPerProduct.get(item.menuProductId) ?? 0;
  return per * item.quantity;
}
```

```typescript
import { query } from "../_generated/server";
import { v } from "convex/values";
import type { Doc, Id } from "../_generated/dataModel";
import {
  getProductionUnitsPerProduct,
  unitsForOrderItem,
} from "./productionUnitHelpers";
import { toDisplayChannel, type DisplayChannel } from "./channelTaxonomy";

/** Shared filter arg validator for all unitEconomics queries. */
const filterArgs = {
  fromTs: v.number(), // inclusive, ms since epoch
  toTs: v.number(),   // exclusive, ms since epoch
  channels: v.optional(v.array(v.string())), // DisplayChannel names; undefined = all
  menuProductIds: v.optional(v.array(v.id("menuProducts"))), // undefined = all
};

type FilterArgs = {
  fromTs: number;
  toTs: number;
  channels?: string[];
  menuProductIds?: Id<"menuProducts">[];
};

/** Shared loader: returns filtered orders + items + unit-per-product map. */
async function loadFilteredData(
  ctx: Parameters<typeof getProductionUnitsPerProduct>[0],
  args: FilterArgs,
) {
  const channelSet = args.channels && args.channels.length > 0
    ? new Set(args.channels)
    : null;
  const productSet = args.menuProductIds && args.menuProductIds.length > 0
    ? new Set(args.menuProductIds.map((id) => id as string))
    : null;

  const allOrders = await ctx.db.query("orders").collect();
  const orders: Doc<"orders">[] = [];
  for (const o of allOrders) {
    if (o.status === "Draft" || o.status === "Cancelled") continue;
    const ts = o.completedAt ?? o.orderDate;
    if (ts < args.fromTs || ts >= args.toTs) continue;
    if (channelSet && !channelSet.has(toDisplayChannel(o.channel))) continue;
    orders.push(o);
  }

  const orderById = new Map<string, Doc<"orders">>();
  for (const o of orders) orderById.set(o._id as string, o);

  const allItems = await ctx.db.query("orderItems").collect();
  const items: Doc<"orderItems">[] = [];
  for (const it of allItems) {
    if (it.isCancelled) continue;
    if (!orderById.has(it.orderId as string)) continue;
    if (productSet && (!it.menuProductId || !productSet.has(it.menuProductId as string))) continue;
    items.push(it);
  }

  const unitsPerProduct = await getProductionUnitsPerProduct(ctx);

  return { orders, items, orderById, unitsPerProduct };
}

/** Width of the prior-period comparison window. */
function priorPeriod(args: FilterArgs): FilterArgs {
  const span = args.toTs - args.fromTs;
  return { ...args, fromTs: args.fromTs - span, toTs: args.fromTs };
}

function computeKpis(
  orders: Doc<"orders">[],
  items: Doc<"orderItems">[],
  unitsPerProduct: Map<Id<"menuProducts">, number>,
) {
  let grossRevenue = 0;
  let discount = 0;
  let units = 0;
  for (const it of items) {
    grossRevenue += it.quantity * it.unitPrice;
    discount += it.discountAmount ?? 0;
    units += unitsForOrderItem(it, unitsPerProduct);
  }
  const netRevenue = grossRevenue - discount; // v1: no platform fees (deferred)
  const orderCount = orders.length;
  return {
    grossRevenue,
    netRevenue,
    units,
    orderCount,
    aovGross: orderCount === 0 ? 0 : grossRevenue / orderCount,
    aovNet: orderCount === 0 ? 0 : netRevenue / orderCount,
    revPerUnit: units === 0 ? 0 : netRevenue / units,
    unitsPerTxn: orderCount === 0 ? 0 : units / orderCount,
  };
}

function deltaPct(current: number, prior: number): number | null {
  if (prior === 0) return null;
  return ((current - prior) / prior) * 100;
}

export const kpiSummary = query({
  args: filterArgs,
  handler: async (ctx, args) => {
    const current = await loadFilteredData(ctx, args);
    const prior = await loadFilteredData(ctx, priorPeriod(args));
    const cur = computeKpis(current.orders, current.items, current.unitsPerProduct);
    const pri = computeKpis(prior.orders, prior.items, prior.unitsPerProduct);
    return {
      current: cur,
      prior: pri,
      delta: {
        netRevenue: deltaPct(cur.netRevenue, pri.netRevenue),
        units: deltaPct(cur.units, pri.units),
        aovNet: deltaPct(cur.aovNet, pri.aovNet),
        revPerUnit: deltaPct(cur.revPerUnit, pri.revPerUnit),
        orderCount: deltaPct(cur.orderCount, pri.orderCount),
        unitsPerTxn: deltaPct(cur.unitsPerTxn, pri.unitsPerTxn),
      },
    };
  },
});
```

/** Jakarta-local weekday index: 0=Sun..6=Sat per Date.getDay() */
function getJakartaDate(ts: number): Date {
  return new Date(ts + 7 * 60 * 60 * 1000);
}

export const byWeekday = query({
  args: filterArgs,
  handler: async (ctx, args) => {
    const { orders, items, orderById, unitsPerProduct } = await loadFilteredData(ctx, args);
    // Mon..Sun order — index 0 = Mon
    const weekdayIndex = (dow: number) => (dow + 6) % 7;
    const orderCountByDow = new Array(7).fill(0);
    const unitCountByDow = new Array(7).fill(0);
    for (const o of orders) {
      const ts = o.completedAt ?? o.orderDate;
      const idx = weekdayIndex(getJakartaDate(ts).getUTCDay());
      orderCountByDow[idx] += 1;
    }
    for (const it of items) {
      const o = orderById.get(it.orderId as string);
      if (!o) continue;
      const ts = o.completedAt ?? o.orderDate;
      const idx = weekdayIndex(getJakartaDate(ts).getUTCDay());
      unitCountByDow[idx] += unitsForOrderItem(it, unitsPerProduct);
    }
    return {
      labels: ["Mon", "Tue", "Wed", "Thu", "Fri", "Sat", "Sun"],
      orders: orderCountByDow,
      units: unitCountByDow,
    };
  },
});

export const dayHourHeatmap = query({
  args: filterArgs,
  handler: async (ctx, args) => {
    const { orders, items, orderById } = await loadFilteredData(ctx, args);
    // rows: 7 (Mon..Sun), cols: 8 (0-3,3-6,6-9,9-12,12-15,15-18,18-21,21-24)
    const grid: number[][] = Array.from({ length: 7 }, () => new Array(8).fill(0));
    const weekdayIndex = (dow: number) => (dow + 6) % 7;
    const binIndex = (hour: number) => Math.min(7, Math.floor(hour / 3));
    for (const it of items) {
      const o = orderById.get(it.orderId as string);
      if (!o) continue;
      const ts = o.completedAt ?? o.orderDate;
      const jd = getJakartaDate(ts);
      const row = weekdayIndex(jd.getUTCDay());
      const col = binIndex(jd.getUTCHours());
      grid[row][col] += it.quantity * it.unitPrice - (it.discountAmount ?? 0);
    }
    const flat = grid.flat();
    const max = flat.reduce((a, b) => Math.max(a, b), 0);
    return {
      grid,
      max,
      rowLabels: ["Mon", "Tue", "Wed", "Thu", "Fri", "Sat", "Sun"],
      colLabels: ["0-3", "3-6", "6-9", "9-12", "12-15", "15-18", "18-21", "21-24"],
    };
  },
});

export const channelEconomics = query({
  args: filterArgs,
  handler: async (ctx, args) => {
    const { orders, items, orderById, unitsPerProduct } = await loadFilteredData(ctx, args);

    const byChannel = new Map<DisplayChannel, {
      gross: number; discount: number; units: number; orders: Set<string>;
    }>();
    function bucket(ch: DisplayChannel) {
      if (!byChannel.has(ch)) {
        byChannel.set(ch, { gross: 0, discount: 0, units: 0, orders: new Set() });
      }
      return byChannel.get(ch)!;
    }

    for (const it of items) {
      const o = orderById.get(it.orderId as string);
      if (!o) continue;
      const ch = toDisplayChannel(o.channel);
      const b = bucket(ch);
      b.gross += it.quantity * it.unitPrice;
      b.discount += it.discountAmount ?? 0;
      b.units += unitsForOrderItem(it, unitsPerProduct);
      b.orders.add(o._id as string);
    }

    const rows = Array.from(byChannel.entries()).map(([channel, b]) => {
      const net = b.gross - b.discount;
      // v1: platform fees = 0 (deferred per spec §2/§5.4). Take-rate reflects discount depth only for now.
      const fees = 0;
      return {
        channel,
        gross: b.gross,
        discount: b.discount,
        fees,
        net,
        units: b.units,
        orderCount: b.orders.size,
        takePct: b.gross === 0 ? 0 : (b.discount + fees) / b.gross * 100,
        revPerUnit: b.units === 0 ? 0 : net / b.units,
        netPerUnit: b.units === 0 ? 0 : net / b.units,
      };
    });
    rows.sort((a, b) => b.gross - a.gross);
    return rows;
  },
});
```

/** Bucket a Jakarta timestamp into a YYYY-MM-DD or ISO-week key. */
function bucketKey(ts: number, granularity: "day" | "week"): string {
  const d = getJakartaDate(ts);
  if (granularity === "day") {
    return d.toISOString().split("T")[0];
  }
  // ISO week: yyyy-Www (Monday start). Simple implementation.
  const tmp = new Date(Date.UTC(d.getUTCFullYear(), d.getUTCMonth(), d.getUTCDate()));
  const day = tmp.getUTCDay() || 7;
  tmp.setUTCDate(tmp.getUTCDate() + 4 - day);
  const yearStart = new Date(Date.UTC(tmp.getUTCFullYear(), 0, 1));
  const week = Math.ceil((((tmp.getTime() - yearStart.getTime()) / 86400000) + 1) / 7);
  return `${tmp.getUTCFullYear()}-W${String(week).padStart(2, "0")}`;
}

export const volumeByType = query({
  args: { ...filterArgs, granularity: v.union(v.literal("day"), v.literal("week")) },
  handler: async (ctx, args) => {
    const { items, orderById } = await loadFilteredData(ctx, args);
    const { byProduct, typeCodes, typeCodeToName } = await getProductionUnitsByTypePerProduct(ctx);

    const bucketMap = new Map<string, Map<string, number>>(); // bucket -> typeCode -> units
    for (const it of items) {
      if (!it.menuProductId) continue;
      const o = orderById.get(it.orderId as string);
      if (!o) continue;
      const ts = o.completedAt ?? o.orderDate;
      const key = bucketKey(ts, args.granularity);
      if (!bucketMap.has(key)) bucketMap.set(key, new Map());
      const bucket = bucketMap.get(key)!;
      const perType = byProduct.get(it.menuProductId);
      if (!perType) continue;
      for (const [code, pcsPerUnit] of perType.entries()) {
        bucket.set(code, (bucket.get(code) ?? 0) + pcsPerUnit * it.quantity);
      }
    }

    const sortedBuckets = Array.from(bucketMap.keys()).sort();
    const series = typeCodes.map((code) => ({
      code,
      name: typeCodeToName.get(code) ?? code,
      values: sortedBuckets.map((k) => bucketMap.get(k)?.get(code) ?? 0),
    }));
    return { buckets: sortedBuckets, series };
  },
});
```

    const { items, orderById } = await loadFilteredData(ctx, args);

    const cell = new Map<string, Map<DisplayChannel, number>>(); // product -> channel -> revenue
    const channelTotals = new Map<DisplayChannel, number>();

    for (const it of items) {
      const o = orderById.get(it.orderId as string);
      if (!o) continue;
      const name = it.productName;
      const ch = toDisplayChannel(o.channel);
      const rev = it.quantity * it.unitPrice - (it.discountAmount ?? 0);
      productTotals.set(name, (productTotals.get(name) ?? 0) + rev);
      channelTotals.set(ch, (channelTotals.get(ch) ?? 0) + rev);
      if (!cell.has(name)) cell.set(name, new Map());
      const m = cell.get(name)!;
      m.set(ch, (m.get(ch) ?? 0) + rev);
    }

    const topN = args.topN ?? 8;
    const topProducts = Array.from(productTotals.entries())
      .sort((a, b) => b[1] - a[1])
      .slice(0, topN)
      .map(([name]) => name);
    const channels = Array.from(channelTotals.keys()).sort(
      (a, b) => (channelTotals.get(b) ?? 0) - (channelTotals.get(a) ?? 0),
    );

    const matrix = topProducts.map((product) => {
      const channelCells = channels.map((channel) => {
        const rev = cell.get(product)?.get(channel) ?? 0;
        const channelTotal = channelTotals.get(channel) ?? 0;
        return {
          channel,
          revenue: rev,
          pctOfChannel: channelTotal === 0 ? 0 : (rev / channelTotal) * 100,
        };
      });
      return { product, channels: channelCells };
    });
    return { products: topProducts, channels, matrix };
  },
});

export const channelMomentum = query({
  args: filterArgs,
  handler: async (ctx, args) => {
    // Split the window into 6 equal sub-buckets for sparklines
    const bucketCount = 6;
    const span = args.toTs - args.fromTs;
    const bucketSpan = span / bucketCount;
    const buckets: Array<{ fromTs: number; toTs: number }> = [];
    for (let i = 0; i < bucketCount; i++) {
      buckets.push({
        fromTs: args.fromTs + i * bucketSpan,
        toTs: args.fromTs + (i + 1) * bucketSpan,
      });
    }

    const { orders, items, orderById, unitsPerProduct } = await loadFilteredData(ctx, args);
    const priorData = await loadFilteredData(ctx, priorPeriod(args));

    // channel -> { revenue[], units[], aov[] }
    const byChannel = new Map<DisplayChannel, {
      revenue: number[]; units: number[]; orders: number[];
      priorRevenue: number;
    }>();
    function seed(ch: DisplayChannel) {
      if (!byChannel.has(ch)) {
        byChannel.set(ch, {
          revenue: new Array(bucketCount).fill(0),
          units: new Array(bucketCount).fill(0),
          orders: new Array(bucketCount).fill(0),
          priorRevenue: 0,
        });
      }
      return byChannel.get(ch)!;
    }

    for (const o of orders) {
      const ch = toDisplayChannel(o.channel);
      const b = seed(ch);
      const ts = o.completedAt ?? o.orderDate;
      const idx = Math.min(bucketCount - 1, Math.floor((ts - args.fromTs) / bucketSpan));
      b.orders[idx] += 1;
    }
    for (const it of items) {
      const o = orderById.get(it.orderId as string);
      if (!o) continue;
      const ch = toDisplayChannel(o.channel);
      const b = seed(ch);
      const ts = o.completedAt ?? o.orderDate;
      const idx = Math.min(bucketCount - 1, Math.floor((ts - args.fromTs) / bucketSpan));
      b.revenue[idx] += it.quantity * it.unitPrice - (it.discountAmount ?? 0);
      b.units[idx] += unitsForOrderItem(it, unitsPerProduct);
    }
    for (const it of priorData.items) {
      const o = priorData.orderById.get(it.orderId as string);
      if (!o) continue;
      const ch = toDisplayChannel(o.channel);
      const b = seed(ch);
      b.priorRevenue += it.quantity * it.unitPrice - (it.discountAmount ?? 0);
    }

    return Array.from(byChannel.entries()).map(([channel, b]) => {
      const totalRevenue = b.revenue.reduce((a, c) => a + c, 0);
      const wowPct = b.priorRevenue === 0 ? null : ((totalRevenue - b.priorRevenue) / b.priorRevenue) * 100;
      const aov = b.revenue.map((r, i) => (b.orders[i] === 0 ? 0 : r / b.orders[i]));
      return {
        channel,
        revenueSpark: b.revenue,
        unitsSpark: b.units,
        aovSpark: aov,
        totalRevenue,
        priorRevenue: b.priorRevenue,
        wowPct,
      };
    }).sort((a, b) => b.totalRevenue - a.totalRevenue);
  },
});

export const rollingTrend = query({
  args: filterArgs,
  handler: async (ctx, args) => {
    const { items, orderById } = await loadFilteredData(ctx, args);
    const daily = new Map<string, number>();
    for (const it of items) {
      const o = orderById.get(it.orderId as string);
      if (!o) continue;
      const ts = o.completedAt ?? o.orderDate;
      const key = bucketKey(ts, "day");
      const rev = it.quantity * it.unitPrice - (it.discountAmount ?? 0);
      daily.set(key, (daily.get(key) ?? 0) + rev);
    }
    const sortedDates = Array.from(daily.keys()).sort();
    const dailyValues = sortedDates.map((d) => daily.get(d) ?? 0);

    function rolling(window: number): number[] {
      return dailyValues.map((_, i) => {
        const start = Math.max(0, i - window + 1);
        const slice = dailyValues.slice(start, i + 1);
        return slice.reduce((a, b) => a + b, 0) / slice.length;
      });
    }

    return {
      dates: sortedDates,
      daily: dailyValues,
      rolling7: rolling(7),
      rolling28: rolling(28),
    };
  },
});

export function AnalyticsFilterProvider({ children }: { children: ReactNode }) {
  const [params, setParams] = useSearchParams();

  const filters = useMemo<AnalyticsFilters>(() => {
    const now = Date.now();
    const defaultFrom = now - 30 * 86400000;
    const fromTs = Number(params.get("from")) || defaultFrom;
    const toTs = Number(params.get("to")) || now;
    const channels = (params.get("channels") ?? "")
      .split(",")
      .filter(Boolean) as DisplayChannel[];
    const menuProductIds = (params.get("products") ?? "")
      .split(",")
      .filter(Boolean) as Id<"menuProducts">[];
    return { fromTs, toTs, channels, menuProductIds };
  }, [params]);

  const setFilters = (next: Partial<AnalyticsFilters>) => {
    const merged = { ...filters, ...next };
    const p = new URLSearchParams(params);
    p.set("from", String(merged.fromTs));
    p.set("to", String(merged.toTs));
    if (merged.channels.length) p.set("channels", merged.channels.join(","));
    else p.delete("channels");
    if (merged.menuProductIds.length) p.set("products", merged.menuProductIds.join(","));
    else p.delete("products");
    setParams(p, { replace: true });
  };

  return (
    <AnalyticsFilterContext.Provider value={{ filters, setFilters }}>
      {children}
    </AnalyticsFilterContext.Provider>
  );
}

export function useAnalyticsFilters() {
  const ctx = useContext(AnalyticsFilterContext);
  if (!ctx) throw new Error("useAnalyticsFilters must be used inside AnalyticsFilterProvider");
  return ctx;
}
