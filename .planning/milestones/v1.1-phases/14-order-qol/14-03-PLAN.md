    **New query: `listForKanban` (convex/orders/queries.ts):**
    Returns orders grouped into 6 Kanban columns, each sorted by dueDate ascending.

    ```typescript
    export const listForKanban = query({
      args: {},
      handler: async (ctx) => {
        // Define column -> status mapping
        const columns = [
          { key: "draft", statuses: ["Draft"] },
          { key: "awaiting_payment", statuses: ["AwaitingPayment"] },
          { key: "payment_received", statuses: ["PaymentReceived"] },
          { key: "being_prepared", statuses: ["BeingPrepared"] },
          { key: "awaiting_delivery", statuses: ["AwaitingDelivery"] },
          { key: "complete", statuses: ["Complete", "Cancelled"] },
        ];

        const result: Record<string, OrderWithItems[]> = {};
        for (const col of columns) {
          const orders = [];
          for (const status of col.statuses) {
            const statusOrders = await ctx.db
              .query("orders")
              .withIndex("by_status_due_date", q => q.eq("status", status))
              .collect();
            orders.push(...statusOrders);
          }
          // Sort by dueDate ascending (nearest due first)
          orders.sort((a, b) => (a.dueDate ?? Infinity) - (b.dueDate ?? Infinity));

          // Enrich with items + customer + creator
          const enriched = await Promise.all(orders.map(async (order) => {
            const items = await ctx.db
              .query("orderItems")
              .withIndex("by_order", q => q.eq("orderId", order._id))
              .filter(q => q.neq(q.field("isCancelled"), true))
              .collect();
            // Resolve creator name from users table
            let creatorName = order.createdBy;
            if (order.createdByUserId) {
              const user = await ctx.db.get(order.createdByUserId);
              if (user) creatorName = user.name;
            }
            return { ...order, items, creatorName };
          }));

          result[col.key] = enriched;
        }
        return result;
      },
    });
    ```

    For the "complete" column: Cancelled orders are included but frontend toggles their visibility. Limit complete column to last 50 orders (performance) using `.take(50)` after sorting by completedAt descending.

    **New query: `getAuditTrail` (convex/orders/queries.ts):**
    ```typescript
    export const getAuditTrail = query({
      args: { orderId: v.id("orders") },
      handler: async (ctx, args) => {
        const events = await ctx.db
          .query("orderEvents")
          .withIndex("by_order", q => q.eq("orderId", args.orderId))
          .collect();
        // Sort newest first
        events.sort((a, b) => b.timestamp - a.timestamp);
        // Enrich with user names
        const enriched = await Promise.all(events.map(async (event) => {
          let userName: string | undefined;
          if (event.userId) {
            const user = await ctx.db.get(event.userId);
            if (user) userName = user.name;
          }
          return { ...event, userName };
        }));
        return enriched;
      },
    });
    ```

    **New query: `searchCustomers` (convex/customers/queries.ts):**
    Add a search query for customer autocomplete:
    ```typescript
    export const search = query({
      args: { query: v.string() },
      handler: async (ctx, args) => {
        if (args.query.length < 2) return [];
        const all = await ctx.db.query("customers").collect();
        const q = args.query.toLowerCase();
        return all
          .filter(c => c.name.toLowerCase().includes(q) || (c.phone && c.phone.includes(q)))
          .slice(0, 10);
      },
    });
    ```
    Note: Convex doesn't have LIKE/full-text on all plans, so collect + filter is the pragmatic approach for the customer table (likely <1000 rows). If a search index exists, use it instead.

      args: {
        orderId: v.id("orders"),
        targetStatus: v.string(), // Which status to revert to
        reason: v.optional(v.string()), // Reason from confirmation modal
        token: v.optional(v.string()),
