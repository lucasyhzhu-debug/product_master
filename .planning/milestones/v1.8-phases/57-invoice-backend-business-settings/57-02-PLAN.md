**2. src/hooks/convex/useBusinessSettings.ts -- Business settings hooks:**

Follow the pattern from useBankAccounts.ts. Import `useSessionQuery` from `convex-helpers/react/sessions`, `useSessionMutation`, `api`, and `createMutationHook`.

```typescript
// Query hooks
export function useBusinessSettings() {
  return useSessionQuery(api.businessSettings.queries.get, {});
}

// Mutation hooks
export const useUpsertBusinessSettings = createMutationHook(
  api.businessSettings.mutations.upsert,
  { successMessage: "Settings saved", errorMessage: "Failed to save settings" }
);

// Upload URL -- raw useSessionMutation (no toast), same pattern as useExpenseUploadUrl
export function useBusinessSettingsUploadUrl() {
  return useSessionMutation(api.businessSettings.mutations.generateUploadUrl);
}

// Types
export type BusinessSettings = NonNullable<ReturnType<typeof useBusinessSettings>>;
```

**3. src/hooks/convex/useInvoice.ts -- Invoice hooks:**

Follow same pattern. This hook is created now but consumed primarily by Phase 58.

```typescript
// Query hooks
export function useInvoicesByOrder(orderId: Id<"orders"> | undefined) {
  return useSessionQuery(
    api.invoices.queries.getByOrder,
    orderId ? { orderId } : "skip"
  );
}

export function useInvoice(invoiceId: Id<"invoices"> | undefined) {
  return useSessionQuery(
    api.invoices.queries.getById,
    invoiceId ? { invoiceId } : "skip"
  );
}

// Mutation hooks
export const useCreateInvoiceDraft = createMutationHook(
  api.invoices.mutations.createDraft,
  { successMessage: "", errorMessage: "Failed to create invoice draft" }
);

export const useUpdateInvoiceDraft = createMutationHook(
  api.invoices.mutations.updateDraft,
  { successMessage: "", errorMessage: "Failed to save invoice" }
);

export const useDiscardInvoiceDraft = createMutationHook(
  api.invoices.mutations.discardDraft,
  { successMessage: "Draft discarded", errorMessage: "Failed to discard draft" }
);

export const useFinalizeInvoice = createMutationHook(
  api.invoices.mutations.finalize,
  { successMessage: "Invoice finalized", errorMessage: "Failed to finalize invoice" }
);

// Types
export type Invoice = NonNullable<ReturnType<typeof useInvoice>>;
```

Note: createDraft and updateDraft have empty successMessage (no toast -- auto-save feedback handled in Phase 58's UI, not via toasts per user decision).

**1. src/components/settings/LogoUploader.tsx:**

Click-to-upload button component for the company logo. Props: `logoUrl: string | null`, `logoStorageId: string | null`, `onUpload: (storageId: Id<"_storage"> | undefined) => void`, `generateUploadUrl: () => Promise<string>`.

NOTE: The `onUpload` prop type accepts `Id<"_storage"> | undefined` because the "Remove Logo" button calls `onUpload(undefined)` to clear the logo.

- Display current logo at fixed height (h-16 / 64px) if `logoUrl` exists, else show placeholder icon (Building2).
- "Upload Logo" button (or "Change Logo" if exists). On click: create hidden `<input type="file" accept="image/png,image/jpeg,image/svg+xml">` and trigger click.
- On file select: validate size <= 1MB (1024*1024 bytes), show toast error if too large. Call `generateUploadUrl()`, then `fetch(url, { method: "POST", headers: { "Content-Type": file.type }, body: file })`. Parse response JSON: `const { storageId } = await response.json();` (Convex returns `{ storageId: string }`). Call `onUpload(storageId)`.
- Show loading spinner during upload.
- "Remove Logo" button (only if logo exists) -- calls `onUpload(undefined)` to clear.
