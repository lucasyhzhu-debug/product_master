```typescript
import { v } from "convex/values";
import { query } from "../_generated/server";
import { requireRole } from "../lib/auth";

/**
 * Phase 74.5.2 Plan 07 — Per-settlement item breakdown query.
 *
 * Returns externalRevenueItems joined to the settlement's linkedRevenueId, enriched
 * with menuProduct display name. Admin + manager gated (financial detail exposure).
 *
 * Pre-74.5.1 settlements return [] — no items were emitted before the saveRevenueItems
 * refactor. UI handles this as empty-state, not an error.
 */
export const getSettlementItems = query({
  args: {
    settlementId: v.id("consignmentSettlements"),
    token: v.string(),
  },
  handler: async (ctx, args) => {
    await requireRole(ctx, args.token, ["admin", "manager"]);
    const settlement = await ctx.db.get(args.settlementId);
    if (!settlement || !settlement.linkedRevenueId) return [];

    const items = await ctx.db
      .query("externalRevenueItems")
      .withIndex("by_revenue", (q) => q.eq("revenueId", settlement.linkedRevenueId!))
      .collect();

    // Enrich each item with menuProduct name for display.
    const enriched = await Promise.all(
      items.map(async (item) => ({
        ...item,
        menuProduct: item.linkedMenuProductId
          ? await ctx.db.get(item.linkedMenuProductId)
          : null,
      })),
    );
    return enriched;
  },
});
```
