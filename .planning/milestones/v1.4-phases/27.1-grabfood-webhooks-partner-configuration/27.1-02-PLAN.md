    ```typescript
    export const updateProductMappingFields = mutation({
      args: {
        token: v.string(),
        mappingId: v.id("externalProductMappings"),
        grabfoodPrice: v.optional(v.number()),
        isAvailable: v.optional(v.boolean()),
      },
      handler: async (ctx, args) => {
        const user = await requireRole(ctx, args.token, ["admin"]);
        const mapping = await ctx.db.get(args.mappingId);
        if (!mapping) throw new Error("Mapping not found");

        const updates: Record<string, any> = {};
        if (args.grabfoodPrice !== undefined) updates.grabfoodPrice = args.grabfoodPrice;
        if (args.isAvailable !== undefined) updates.isAvailable = args.isAvailable;

        if (Object.keys(updates).length > 0) {
          await ctx.db.patch(args.mappingId, updates);
        }
      },
    });
    ```

    **IMPORTANT:** Use `const user = await requireRole(...)` (capture return value) consistent with the pattern fixed in Plan 01.

    **Add `getLatestWebhookError` query in `convex/externalData/queries.ts`:**

    This query powers the sync error banner in the Webhooks tab. It returns the most recent webhook sync log entry with `status: "error"` for a given source, or null if none recent.

    ```typescript
    export const getLatestWebhookError = query({
      args: { source: v.string() },
      handler: async (ctx, args) => {
        // Get the last 10 webhook sync logs for this source, check for recent errors
        const logs = await ctx.db
          .query("externalSyncLogs")
          .withIndex("by_source", (q) => q.eq("source", args.source))
          .order("desc")
          .take(20);

        // Find the most recent webhook error within last 24 hours
        const oneDayAgo = Date.now() - 24 * 60 * 60 * 1000;
        const webhookError = logs.find(
          (log) =>
            log.syncType === "webhook" &&
            log.status === "error" &&
            log.timestamp > oneDayAgo
        );

        if (!webhookError) return null;

        return {
          timestamp: webhookError.timestamp,
          errorMessage: webhookError.errorMessage ?? "Unknown webhook error",
        };
      },
    });
    ```

    Import `query` from `"../_generated/server"` if not already imported.
