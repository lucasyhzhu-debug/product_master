**File 2: `src/hooks/convex/useExternalData.ts`**

Replace `useConvexRestockOverview` hook (lines 191-194). **Typing:** Do NOT use `unknown` -- use `FunctionReturnType` or `Awaited<ReturnType<typeof fetchAction>>` so downstream components can access typed properties:
```typescript
type RestockOverview = FunctionReturnType<typeof api.externalData.actions.fetchRestockOverview>;

export function useConvexRestockOverview() {
  const [data, setData] = useState<RestockOverview | undefined>(undefined);
  const [isLoading, setIsLoading] = useState(true);
  const fetchAction = useAction(api.externalData.actions.fetchRestockOverview);

  const load = useCallback(async () => {
    setIsLoading(true);
    try {
      const result = await fetchAction({});
      setData(result);
    } catch (error) {
      console.error("Failed to fetch restock overview:", error);
    } finally {
      setIsLoading(false);
    }
  }, [fetchAction]);

  useEffect(() => { load(); }, [load]);

  return { data, isLoading, refresh: load };
}
```
