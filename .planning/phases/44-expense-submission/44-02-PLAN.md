/**
 * Expense hooks for expense submission workflow.
 * Query hooks + factory-generated mutation hooks.

import { useQuery } from "convex/react";
import { useSessionMutation } from "convex-helpers/react/sessions";
import { api } from "../../../convex/_generated/api";
import type { Id } from "../../../convex/_generated/dataModel";
import { createMutationHook } from "./createMutationHook";

// QUERY HOOKS

export function useMyExpenses(status?: string) {
  return useQuery(api.expenses.queries.listMyExpenses, status ? { status } : {});
}

export function useExpense(id: Id<"expenses"> | undefined) {
  return useQuery(api.expenses.queries.getById, id ? { expenseId: id } : "skip");
}

export function useExpenseStatusHistory(expenseId: Id<"expenses"> | undefined) {
  return useQuery(
    api.expenses.queries.getStatusHistory,
    expenseId ? { expenseId } : "skip"
  );
}

// MUTATION HOOKS

export const useCreateExpenseDraft = createMutationHook(
  api.expenses.mutations.createDraft,
  { successMessage: "Draft saved", errorMessage: "Failed to save draft" }
);

export const useUpdateExpenseDraft = createMutationHook(
  api.expenses.mutations.updateDraft,
  { successMessage: "Draft updated", errorMessage: "Failed to update draft" }
);

export const useSubmitExpense = createMutationHook(
  api.expenses.mutations.submitExpense,
  { successMessage: "Expense submitted for approval", errorMessage: "Failed to submit expense" }
);

// Special hook for generateUploadUrl -- needs raw mutation, not toast wrapper
// because it's called as part of the upload flow, not as a standalone action
export function useExpenseUploadUrl() {
  return useSessionMutation(api.expenses.mutations.generateUploadUrl);
}

// TYPES

export type Expense = NonNullable<ReturnType<typeof useMyExpenses>>[number];
export type ExpenseStatusHistoryEntry = NonNullable<
  ReturnType<typeof useExpenseStatusHistory>
>[number];

**4. Create `src/components/expenses/ReceiptUpload.tsx`:**

File input with SHA-256 hashing and upload via Convex storage. Follow the uploadScreenshot pattern from `useFeedback.ts`:

```typescript
import { useState, useCallback } from "react";
import { Button } from "@/components/ui/button";
import { Upload, X, FileImage, Loader2 } from "lucide-react";
import type { Id } from "../../../convex/_generated/dataModel";

interface ReceiptUploadProps {
  onUpload: (result: { storageId: Id<"_storage">; hash: string }) => void;
  onRemove: () => void;
  currentFileId?: Id<"_storage">;
  disabled?: boolean;
}

/** Compute SHA-256 hash of a file using Web Crypto API */
async function computeSha256(file: File): Promise<string> {
  const buffer = await file.arrayBuffer();
  const hashBuffer = await crypto.subtle.digest("SHA-256", buffer);
  const hashArray = Array.from(new Uint8Array(hashBuffer));
  return hashArray.map((b) => b.toString(16).padStart(2, "0")).join("");
}

const MAX_FILE_SIZE = 5 * 1024 * 1024; // 5MB
const ACCEPTED_TYPES = ["image/jpeg", "image/png", "image/webp", "application/pdf"];

export function ReceiptUpload({
  onUpload,
  onRemove,
  currentFileId,
  disabled = false,
}: ReceiptUploadProps) {
  const [uploading, setUploading] = useState(false);
  const [fileName, setFileName] = useState<string | null>(null);
  const [error, setError] = useState<string | null>(null);

  // The actual upload is triggered by the parent component which has access
  // to the generateUploadUrl mutation. This component handles file selection,
  // validation, hashing, and delegates the upload to the parent.

  const handleFileSelect = useCallback(
    async (e: React.ChangeEvent<HTMLInputElement>) => {
      const file = e.target.files?.[0];
      if (!file) return;

      setError(null);

      // Validate file type
      if (!ACCEPTED_TYPES.includes(file.type)) {
        setError("Unsupported file type. Use JPEG, PNG, WebP, or PDF.");
        return;
      }

      // Validate file size
      if (file.size > MAX_FILE_SIZE) {
        setError("File too large. Maximum 5MB.");
        return;
      }

      setUploading(true);
      setFileName(file.name);

      try {
        // Compute SHA-256 hash client-side
        const hash = await computeSha256(file);

        // Parent handles the actual Convex upload
        // We pass the file data through a custom event pattern
        // The parent will call generateUploadUrl, fetch to upload, then call onUpload
        onUpload({ storageId: file as unknown as Id<"_storage">, hash });
        // Note: The parent component must handle the actual upload flow.
        // See ExpenseSubmit.tsx for the full upload orchestration.
      } catch {
        setError("Failed to process receipt");
      } finally {
        setUploading(false);
      }
    },
    [onUpload]
  );

  // ... render file input, preview, remove button
}
```

IMPORTANT: The ReceiptUpload component above is a simplified sketch. The actual implementation needs a cleaner upload flow:

The component should accept `generateUploadUrl` as a prop (the raw mutation function from `useExpenseUploadUrl`). The full flow is:
1. User selects file
2. Component validates size/type
3. Component computes SHA-256 hash
4. Component calls `generateUploadUrl()` to get signed URL
5. Component POSTs file to signed URL, gets back `storageId`
6. Component calls `onUpload({ storageId, hash })` with the real storageId and hash

This keeps the upload logic self-contained in the component. The parent only needs to pass the `generateUploadUrl` mutation and handle the `onUpload` callback to store storageId + hash in form state.

If `currentFileId` is provided, show a "Receipt attached" indicator with a remove button. If no file, show the upload input.
