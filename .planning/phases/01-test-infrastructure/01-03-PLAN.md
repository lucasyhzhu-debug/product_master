**AAA pattern example (from RESEARCH.md):**
```typescript
test('Confirmed → InProduction transition on first ball allocation', async () => {
  const t = convexTest(schema);

  // ARRANGE: Create order at Confirmed status
  const orderId = await createOrderAtStatus(t, { targetStatus: 'Confirmed' });
  let order = await t.run(async (ctx) => ctx.db.get(orderId));
  expect(order?.status).toBe('Confirmed');

  // ACT: Add balls (triggers auto-transition)
  await t.mutation(api.orders.mutations.kitchen.addBallsToTray, {
    ballType: 'big',
    count: 10
  });

  // ASSERT: Order auto-transitioned to InProduction
  order = await t.run(async (ctx) => ctx.db.get(orderId));
  expect(order?.status).toBe('InProduction');
});
```
