Create `convex/employees/queries.ts` with two queries:

**`list` query** — Returns all employees, optionally filtered by `employeeType` and/or `isActive` status. Admin-only.

```typescript
import { v } from "convex/values";
import { protectedQuery } from "../lib/functions";

export const list = protectedQuery({
  roles: ["admin"],
  args: {
    employeeType: v.optional(v.union(v.literal("contractor"), v.literal("staff"))),
    isActive: v.optional(v.boolean()),
  },
  handler: async (ctx, args) => {
    let employees;

    if (args.employeeType !== undefined && args.isActive !== undefined) {
      employees = await ctx.db
        .query("employees")
        .withIndex("by_type_active", (q) =>
          q.eq("employeeType", args.employeeType!).eq("isActive", args.isActive!)
        )
        .order("desc")
        .collect();
    } else if (args.isActive !== undefined) {
      employees = await ctx.db
        .query("employees")
        .withIndex("by_active", (q) => q.eq("isActive", args.isActive!))
        .order("desc")
        .collect();
    } else {
      employees = await ctx.db
        .query("employees")
        .order("desc")
        .collect();
    }

    // Resolve createdBy user names
    const userIds = new Set<string>();
    for (const emp of employees) {
      userIds.add(emp.createdBy as string);
    }

    const userMap = new Map<string, string>();
    const users = await Promise.all(
      [...userIds].map((id) => ctx.db.get(id as any))
    );
    for (const user of users) {
      if (user) {
        userMap.set(user._id as string, (user as any).name ?? "Unknown");
      }
    }

    return employees.map((emp) => ({
      ...emp,
      createdByName: userMap.get(emp.createdBy as string) ?? "Unknown",
    }));
  },
});

export const getById = protectedQuery({
  roles: ["admin"],
  args: {
    employeeId: v.id("employees"),
  },
  handler: async (ctx, args) => {
    const employee = await ctx.db.get(args.employeeId);
    if (!employee) {
      throw new Error("Employee not found");
    }

    const createdByUser = await ctx.db.get(employee.createdBy);
    const createdByName = createdByUser?.name ?? "Unknown";

    // Resolve linked user if present
    let linkedUserName: string | undefined;
    if (employee.userId) {
      const linkedUser = await ctx.db.get(employee.userId);
      linkedUserName = linkedUser?.name ?? undefined;
    }

    return {
      ...employee,
      createdByName,
      linkedUserName,
    };
  },
});
```

Create `convex/employees/mutations.ts` with create, update, and toggleActive mutations:

```typescript
import { v } from "convex/values";
import { protectedMutation } from "../lib/functions";

// ---------------------------------------------------------------------------
// create — create a new employee record
// ---------------------------------------------------------------------------
export const create = protectedMutation({
  roles: ["admin"],
  args: {
    name: v.string(),
    role: v.string(),
    employeeType: v.union(v.literal("contractor"), v.literal("staff")),
    frequency: v.optional(v.union(v.literal("weekly"), v.literal("monthly"))),
    bankName: v.optional(v.string()),
    bankAccountNumber: v.optional(v.string()),
    bankAccountHolder: v.optional(v.string()),
    phone: v.optional(v.string()),
    email: v.optional(v.string()),
    joinDate: v.optional(v.number()),
    monthlySalary: v.optional(v.number()),
    notes: v.optional(v.string()),
    userId: v.optional(v.id("users")),
  },
  handler: async (ctx, args) => {
    if (!args.name.trim()) {
      throw new Error("Employee name is required");
    }
    if (!args.role.trim()) {
      throw new Error("Employee role is required");
    }

    // Validate userId uniqueness if provided
    if (args.userId) {
      const existing = await ctx.db
        .query("employees")
        .withIndex("by_userId", (q) => q.eq("userId", args.userId!))
        .first();
      if (existing) {
        throw new Error("An employee record already exists for this user");
      }
    }

    // Validate monthlySalary if provided
    if (args.monthlySalary !== undefined && args.monthlySalary < 0) {
      throw new Error("Monthly salary cannot be negative");
    }

    const employeeId = await ctx.db.insert("employees", {
      name: args.name.trim(),
      role: args.role.trim(),
      employeeType: args.employeeType,
      frequency: args.frequency,
      bankName: args.bankName?.trim() || undefined,
      bankAccountNumber: args.bankAccountNumber?.trim() || undefined,
      bankAccountHolder: args.bankAccountHolder?.trim() || undefined,
      phone: args.phone?.trim() || undefined,
      email: args.email?.trim() || undefined,
      joinDate: args.joinDate,
      monthlySalary: args.monthlySalary,
      notes: args.notes?.trim() || undefined,
      userId: args.userId,
      isActive: true,
      createdBy: ctx.user._id,
      updatedAt: Date.now(),
    });

    return { employeeId };
  },
});

// ---------------------------------------------------------------------------
// update — update an existing employee record
// ---------------------------------------------------------------------------
export const update = protectedMutation({
  roles: ["admin"],
  args: {
    employeeId: v.id("employees"),
    name: v.string(),
    role: v.string(),
    employeeType: v.union(v.literal("contractor"), v.literal("staff")),
    frequency: v.optional(v.union(v.literal("weekly"), v.literal("monthly"))),
    bankName: v.optional(v.string()),
    bankAccountNumber: v.optional(v.string()),
    bankAccountHolder: v.optional(v.string()),
    phone: v.optional(v.string()),
    email: v.optional(v.string()),
    joinDate: v.optional(v.number()),
    monthlySalary: v.optional(v.number()),
    notes: v.optional(v.string()),
    userId: v.optional(v.id("users")),
  },
  handler: async (ctx, args) => {
    const employee = await ctx.db.get(args.employeeId);
    if (!employee) {
      throw new Error("Employee not found");
    }

    if (!args.name.trim()) {
      throw new Error("Employee name is required");
    }
    if (!args.role.trim()) {
      throw new Error("Employee role is required");
    }

    // Validate userId uniqueness if changing
    if (args.userId && args.userId !== employee.userId) {
      const existing = await ctx.db
        .query("employees")
        .withIndex("by_userId", (q) => q.eq("userId", args.userId!))
        .first();
      if (existing) {
        throw new Error("An employee record already exists for this user");
      }
    }

    if (args.monthlySalary !== undefined && args.monthlySalary < 0) {
      throw new Error("Monthly salary cannot be negative");
    }

    await ctx.db.patch(args.employeeId, {
      name: args.name.trim(),
      role: args.role.trim(),
      employeeType: args.employeeType,
      frequency: args.frequency,
      bankName: args.bankName?.trim() || undefined,
      bankAccountNumber: args.bankAccountNumber?.trim() || undefined,
      bankAccountHolder: args.bankAccountHolder?.trim() || undefined,
      phone: args.phone?.trim() || undefined,
      email: args.email?.trim() || undefined,
      joinDate: args.joinDate,
      monthlySalary: args.monthlySalary,
      notes: args.notes?.trim() || undefined,
      userId: args.userId,
      updatedAt: Date.now(),
    });

    return { success: true };
  },
});

// ---------------------------------------------------------------------------
// toggleActive — soft-delete / reactivate an employee
// ---------------------------------------------------------------------------
export const toggleActive = protectedMutation({
  roles: ["admin"],
  args: {
    employeeId: v.id("employees"),
  },
  handler: async (ctx, args) => {
    const employee = await ctx.db.get(args.employeeId);
    if (!employee) {
      throw new Error("Employee not found");
    }

    await ctx.db.patch(args.employeeId, {
      isActive: !employee.isActive,
      updatedAt: Date.now(),
    });

    return { isActive: !employee.isActive };
  },
});
```
</action>
