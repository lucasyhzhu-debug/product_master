```typescript
/**
 * Employee hooks for employee roster management.
 * Query hooks + factory-generated mutation hooks.
 */
import { useSessionQuery } from "convex-helpers/react/sessions";
import { api } from "../../../convex/_generated/api";
import type { Id } from "../../../convex/_generated/dataModel";
import { createMutationHook } from "./createMutationHook";

// ============================================================================
// QUERY HOOKS
// ============================================================================

/** List employees with optional type and status filters */
export function useEmployees(
  employeeType?: "contractor" | "staff",
  isActive?: boolean,
) {
  return useSessionQuery(api.employees.queries.list, {
    ...(employeeType !== undefined && { employeeType }),
    ...(isActive !== undefined && { isActive }),
  });
}

/** List active employees only (for payroll picker) */
export function useActiveEmployees() {
  return useSessionQuery(api.employees.queries.listActive, {});
}

/** Get a single employee by ID */
export function useEmployee(id: Id<"employees"> | undefined) {
  return useSessionQuery(
    api.employees.queries.getById,
    id ? { employeeId: id } : "skip",
  );
}

/** Get employee by linked userId (for reimbursement bank detail auto-populate) */
export function useEmployeeByUserId(userId: Id<"users"> | undefined) {
  return useSessionQuery(
    api.employees.queries.getByUserId,
    userId ? { userId } : "skip",
  );
}

// ============================================================================
// MUTATION HOOKS
// ============================================================================

export const useCreateEmployee = createMutationHook(
  api.employees.mutations.create,
  { successMessage: "Employee created", errorMessage: "Failed to create employee" },
);

export const useUpdateEmployee = createMutationHook(
  api.employees.mutations.update,
  { successMessage: "Employee updated", errorMessage: "Failed to update employee" },
);

export const useToggleEmployeeActive = createMutationHook(
  api.employees.mutations.toggleActive,
  { successMessage: "Employee status updated", errorMessage: "Failed to update employee status" },
);

// ============================================================================
// TYPES
// ============================================================================

export type Employee = NonNullable<ReturnType<typeof useEmployees>>[number];
```

```typescript
  useActiveEmployees,
  useEmployee,

  useToggleEmployeeActive,
  type Employee,
} from "./useEmployees";
```
