```tsx
import type { LucideIcon } from 'lucide-react';
import { Button } from '@/components/ui/button';
import { Card, CardContent } from '@/components/ui/card';

interface EmptyStateProps {
  icon: LucideIcon;
  title: string;
  description: string;
  action?: {
    label: string;
    onClick: () => void;
  };
}

export function EmptyState({ icon: Icon, title, description, action }: EmptyStateProps) {
  return (
    <Card className="border-dashed">
      <CardContent className="py-16 text-center">
        {/* Larger icon circle: 96px diameter, muted/50 bg, gentler icon */}
        <div className="w-24 h-24 mx-auto mb-6 rounded-full bg-muted/50 flex items-center justify-center">
          <Icon className="h-12 w-12 text-muted-foreground/60" />
        </div>
        <h3 className="text-lg font-semibold mb-2">{title}</h3>
        <p className="text-muted-foreground mb-6 max-w-md mx-auto leading-relaxed">
          {description}
        </p>
        {action && (
          <Button onClick={action.onClick} size="lg">
            {action.label}
          </Button>
        )}
      </CardContent>
    </Card>
  );
}
```

Key changes from current:
- Icon circle: `w-24 h-24` (was `w-16 h-16`) — 96px diameter per research
- Icon size: `h-12 w-12` (was `h-8 w-8`) — 48px
- Icon opacity: `text-muted-foreground/60` (gentler)
- Circle bg: `bg-muted/50` (was `bg-muted`)
- Description: `max-w-md` (was `max-w-sm`), added `leading-relaxed`
- Spacing: `py-16` (was `py-12`), `mb-6` (was `mb-4`)
- Card: `border-dashed` for visual softness
- Button: `size="lg"` for prominent CTA
- Interface unchanged — backward compatible

**IMPORTANT:**
- Keep the EmptyState interface identical (no breaking changes)
- Skeleton components should not import from pages — they're standalone
- Do not change how pages USE loading/empty states (that's page logic)
