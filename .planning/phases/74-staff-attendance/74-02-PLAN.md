    ```tsx
    import { useEffect, useState } from "react";
    import { Clock } from "lucide-react";

    interface RunningTimerProps {
      /** Clock-in epoch ms */
      clockIn: number;
      className?: string;
    }

    /**
     * Minute-resolution running timer. Ticks every 60s (D-01 Discretion: second-level precision overkill).
     * Display only — authoritative hours computed server-side in getStaffPerformanceSummary via durationMs.
     * A kiosk clock manipulation would only inflate the DISPLAY, not the computed hours (T-74-15).
     */
    export function RunningTimer({ clockIn, className }: RunningTimerProps) {
      // Display only — authoritative hours computed server-side in getStaffPerformanceSummary via durationMs
      const [now, setNow] = useState(() => Date.now());
      useEffect(() => {
        const id = setInterval(() => setNow(Date.now()), 60_000);
        return () => clearInterval(id);
      }, []);
      const elapsedMs = Math.max(0, now - clockIn);
      const h = Math.floor(elapsedMs / 3_600_000);
      const m = Math.floor((elapsedMs % 3_600_000) / 60_000);
      return (
        <span className={`inline-flex items-center gap-1.5 tabular-nums ${className ?? ""}`}>
          <Clock className="h-4 w-4" />
          {h}h {m}m
        </span>
      );
    }
    ```

    ```tsx
    import { useState } from "react";
    import { Button } from "@/components/ui/button";
    import { useClockOut } from "@/hooks/convex/useAttendance";
    import { toast } from "sonner";
    import { LogOut } from "lucide-react";
    import type { Id } from "../../../convex/_generated/dataModel";

    interface ClockOutButtonProps {
      attendanceId: Id<"staffAttendance">;
      size?: "sm" | "default" | "lg";
      onClockedOut?: () => void;
    }

    export function ClockOutButton({ attendanceId, size = "sm", onClockedOut }: ClockOutButtonProps) {
      const clockOut = useClockOut();
      const [pending, setPending] = useState(false);
      const handle = async () => {
        setPending(true);
        try {
          await clockOut({ attendanceId });
          toast.success("Clocked out");
          onClockedOut?.();
        } catch (err) {
          toast.error(err instanceof Error ? err.message : "Clock-out failed");
        } finally {
          setPending(false);
        }
      };
      return (
        <Button variant="destructive" size={size} onClick={handle} disabled={pending}>
          <LogOut className="mr-1.5 h-4 w-4" />
          {pending ? "Clocking out…" : "Clock Out"}
        </Button>
      );
    }
    ```

    ```tsx
    import { useEffect, useState } from "react";
    import { useNavigate, useLocation } from "react-router-dom";
    import { Card, CardContent, CardHeader, CardTitle } from "@/components/ui/card";
    import { Button } from "@/components/ui/button";
    import { Clock, AlertTriangle, ChefHat, LogOut } from "lucide-react";
    import { toast } from "sonner";
    import { useAuth } from "@/contexts/AuthContext";
    import { useClockIn, useCurrentOpenShift, useMyLastShiftSummary } from "@/hooks/convex/useAttendance";
    import { utcToWibTimeStr, utcToWibDateStr, formatIndonesianDate } from "@/lib/dateUtils";

    export function ClockInGate() {
      const navigate = useNavigate();
      const { user, logout } = useAuth();
      const openShift = useCurrentOpenShift();
      const lastShift = useMyLastShiftSummary();
      const clockIn = useClockIn();
      const [pending, setPending] = useState(false);
      const [now, setNow] = useState(() => Date.now());

      // Minute-resolution live clock
      useEffect(() => {
        const id = setInterval(() => setNow(Date.now()), 60_000);
        return () => clearInterval(id);
      }, []);

      // Auto-redirect if already clocked in today
      // T-74-15: skip auto-redirect when arriving from a permission-denied fallback to avoid flash-loop.
      const location = useLocation();
      const fromPermissionDenied = (location.state as { fromPermissionDenied?: boolean } | null)?.fromPermissionDenied === true;
      const todayWib = utcToWibDateStr(now);
      useEffect(() => {
        if (fromPermissionDenied) return;
        if (openShift && openShift.date === todayWib && !openShift.deletedAt) {
          const timer = setTimeout(() => navigate("/kitchen", { replace: true }), 1000);
          return () => clearTimeout(timer);
        }
      }, [openShift, todayWib, navigate, fromPermissionDenied]);

      const isPriorDayOpen = openShift && openShift.date < todayWib && !openShift.deletedAt;

      const handleClockIn = async () => {
        setPending(true);
        try {
          await clockIn({});
          toast.success("Clocked in");
          navigate("/kitchen", { replace: true });
        } catch (err) {
          toast.error(err instanceof Error ? err.message : "Clock-in failed");
        } finally {
          setPending(false);
        }
      };

      // Loading state
      if (openShift === undefined) {
        return <div className="flex min-h-screen items-center justify-center">Loading…</div>;
      }

      return (
        <div className="flex min-h-[calc(100vh-56px)] flex-col items-center justify-center gap-6 bg-gradient-to-b from-background to-muted/30 p-6">
          <div className="w-full max-w-md space-y-4">
            <div className="text-center">
              <ChefHat className="mx-auto h-12 w-12 text-primary" />
              <h1 className="mt-2 text-2xl font-bold">Welcome, {user?.name ?? "Chef"}</h1>
              <p className="text-muted-foreground tabular-nums">
                {formatIndonesianDate(new Date(now + 7 * 60 * 60 * 1000))} · {utcToWibTimeStr(now)} WIB
              </p>
            </div>

            {isPriorDayOpen ? (
              <Card className="border-destructive">
                <CardHeader>
                  <CardTitle className="flex items-center gap-2 text-destructive">
                    <AlertTriangle className="h-5 w-5" />
                    Open shift from {openShift!.date}
                  </CardTitle>
                </CardHeader>
                <CardContent className="space-y-3">
                  <p className="text-sm text-muted-foreground">
                    You have an open shift from {openShift!.date}. Please ask a manager to correct it before clocking in again.
                  </p>
                  <Button variant="outline" onClick={logout} className="w-full">
                    <LogOut className="mr-2 h-4 w-4" /> Log out
                  </Button>
                </CardContent>
              </Card>
            ) : openShift ? (
              <Card>
                <CardContent className="p-6 text-center">
                  <p className="text-lg">You're clocked in since {utcToWibTimeStr(openShift.clockIn)}. Loading kitchen…</p>
                </CardContent>
              </Card>
            ) : (
              <>
                <Button
                  size="lg"
                  className="h-32 w-full text-2xl font-bold"
                  onClick={handleClockIn}
                  disabled={pending}
                >
                  <Clock className="mr-3 h-8 w-8" />
                  {pending ? "Clocking in…" : "Clock In"}
                </Button>
                {lastShift && (
                  <Card>
                    <CardHeader>
                      <CardTitle className="text-base">Last shift</CardTitle>
                    </CardHeader>
                    <CardContent className="text-sm text-muted-foreground">
                      {lastShift.date} · {Math.floor(lastShift.durationMs / 3_600_000)}h {Math.floor((lastShift.durationMs % 3_600_000) / 60_000)}m
                      {lastShift.ballsProduced > 0 && ` · ${lastShift.ballsProduced} balls`}
                    </CardContent>
                  </Card>
                )}
              </>
            )}
          </div>
        </div>
      );
    }
    ```

## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| Browser → Convex (reactive query) | Staff device may be kiosk-shared — session token scopes identity |
| Gate screen → clockIn mutation | Single tap after PIN login — no re-auth (D-02) |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-74-01 | Spoofing | Gate screen → clockIn | mitigate (defense-in-depth with backend) | `useProtectedMutation` auto-injects token; backend `requireRole` re-validates + derives userId from session. Frontend never passes a userId on clockIn. |
| T-74-03 | Info disclosure | `useMyLastShiftSummary`, `useCurrentOpenShift` | mitigate | Queries scope to `user._id` server-side via `requireRole`. Client cannot request another user's data. |
| T-74-04 | Tampering | D-04 prior-day open shift | mitigate | Gate screen reads `getCurrentOpenShift` and blocks the Clock-In button (hides it, shows block state) when `openShift.date < todayWib`. Backend also throws if a client bypasses the UI guard. |
| T-74-07 | DoS (client) | Kiosk left on gate screen forever with running timer interval | accept | `setInterval` fires once per minute — negligible CPU. No network traffic (Convex reactive query only re-fires on data change). |
| T-74-08 | Tampering | D-08 nudge shows for wrong user if kiosk session swaps | mitigate | `useCurrentOpenShift` is reactive on the current session's userId; when a new user logs in, `useAuth` provides a new token, queries re-fire, nudge dialog dismisses (attendanceId becomes the new user's open shift or null). |
| T-74-15 | UX / Denial-of-service | ClockInGate auto-redirect + ProtectedRoute fallback | mitigate | When a kitchen user hits an unauthorized URL, ProtectedRoute redirects to `/kitchen/clock`. ClockInGate's auto-redirect-to-`/kitchen`-when-clocked-in would create a flash-loop. Mitigation: ClockInGate reads `location.state?.fromPermissionDenied`; when true, suppresses the auto-redirect so the user sees the gate screen explicitly. ProtectedRoute must set this flag when redirecting. |
| T-74-16 | Tampering (client clock) | RunningTimer kiosk clock manipulation | accept (display-only) | RunningTimer uses `Date.now()` on the client — a user could manipulate the kiosk clock to inflate the running display. Authoritative hours are computed server-side in `getStaffPerformanceSummary` via `durationMs`. Display drift is not fraud. Documented in a code comment ("Display only — authoritative hours computed server-side..."). |
</threat_model>
