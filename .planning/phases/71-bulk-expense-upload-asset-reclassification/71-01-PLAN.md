    - `importBatchId: v.string()` -- client-generated UUID for batch traceability
    - `rows: v.array(v.object({...}))` -- array of row objects with fields:
      - `date: v.number()` -- epoch ms
      - `amount: v.number()` -- positive integer IDR
      - `description: v.string()`
      - `vendorName: v.string()` -- empty string if not provided
      - `accountId: v.id("accounts")` -- resolved on client from account name match

      - `paymentMethod: v.union(v.literal("employee_paid"), v.literal("company_paid"), v.literal("payment_request"))`
      - `receiptUrl: v.optional(v.string())`
      - `trusted: v.boolean()` -- auto-approve flag

    Handler logic:
    1. Validate batch size: `if (args.rows.length > 50) throw new ConvexError("Maximum 50 rows per batch")`
    2. Check if ANY row has `trusted: true` -- if so, verify `ctx.user.role` is in `APPROVER_ROLES`. If not, throw `new ConvexError("Only managers and admins can auto-approve expenses")` (per D-08)
    3. Resolve Cash account once: `const cashAccount = await resolveAccount(ctx, "1100")`
    4. Loop through each row:
       a. `validateExpenseAmount(row.amount)` -- throws if not positive integer
       b. `const expenseNumber = await getNextNumber(ctx, "EXP")`
       c. Determine status: `row.trusted ? "recorded" : "submitted"` 
       d. Insert expense record with ALL required fields:
          ```typescript
          const expenseId = await ctx.db.insert("expenses", {
            expenseNumber,
            submittedBy: row.submitterId,
            amount: row.amount,
            accountId: row.accountId,
            expenseDate: row.date,
            description: row.description,
            vendorName: row.vendorName,
            paymentMethod: row.paymentMethod,
            status,
            lateSubmission: false,
            submittedAt: Date.now(),
            importBatchId: args.importBatchId,
            createdAt: Date.now(),
          });
          ```
       e. Record audit trail: `await recordStatusChange(ctx, expenseId, undefined, status, ctx.user._id, "Bulk CSV import")`
       f. If trusted: create JE via `createJournalEntryWithLines`:
          - date: `row.date`
          - description: `"[Bulk Import] ${row.description}"`
          - sourceType: `"expense_approval"` (reuses existing type -- consistent with how approved expenses create JEs)
          - sourceId: `expenseId as string`
          - createdBy: `ctx.user._id`
          - lines: `[buildDebitLine(row.accountId, row.amount, row.description), buildCreditLine(cashAccount._id, row.amount)]`
          - Then patch expense with journalEntryId: `await ctx.db.patch(expenseId, { journalEntryId: jeId, approvedBy: ctx.user._id, approvedAt: Date.now() })`
    5. Return `{ created: args.rows.length, autoApproved: trustedCount, submitted: untrustedCount }`

    **Important:** Do NOT create rows in parallel within the handler. Sequential loop is required because Convex transactions are serializable and `getNextNumber` must be called sequentially.

    if (args.disposalType === "reclassify_to_expense") {
      // Validate submitterId is provided
      if (!args.submitterId) {
        throw new ConvexError("Expense owner (submitterId) is required for asset reclassification");
      }

      const nbv = asset.cost - asset.accumulatedDepreciation;
      if (nbv <= 0) {
        throw new ConvexError("Cannot reclassify: Net Book Value must be positive");
      }

      // Import needed functions
      const { getNextNumber } = await import("../lib/counter");
      // NO -- do NOT use dynamic import in Convex (Pitfall 8 from CLAUDE.md)
      // Instead, add static import at top of file

      // Resolve target expense account
      let targetExpenseAccountId: Id<"accounts">;
      if (args.targetExpenseAccountId) {
        targetExpenseAccountId = args.targetExpenseAccountId;
      } else {
        const defaultCode = getReclassificationExpenseCode(asset.category);
        const targetAccount = await resolveAccount(ctx, defaultCode);
        targetExpenseAccountId = targetAccount._id;
      }

      // Resolve GL accounts for JE
      const cat = ASSET_CATEGORIES.find((c) => c.key === asset.category);
      const assetAccountCode = cat ? getAssetAccountCode(cat) : "1500";
      const fixedAssetAccount = await resolveAccount(ctx, assetAccountCode);
      const accumAccount = cat?.glAccumCode ? await resolveAccount(ctx, cat.glAccumCode) : null;

      // Build reclassification JE lines (per D-12)
      const jeLines = [];

      // DR Target Expense Account (NBV)
      jeLines.push(buildDebitLine(targetExpenseAccountId, nbv, `Reclassified: ${asset.name}`));

      // DR Accumulated Depreciation (full accumulated amount, if > 0)
      if (asset.accumulatedDepreciation > 0 && accumAccount) {
        jeLines.push(buildDebitLine(accumAccount._id, asset.accumulatedDepreciation, "Remove accumulated depreciation"));
      }

      // CR Fixed Assets (original cost)
      jeLines.push(buildCreditLine(fixedAssetAccount._id, asset.cost, "Remove asset cost"));

      // Create JE
      const jeId = await createJournalEntryWithLines(ctx, {
        date: args.disposalDate,
        description: `Asset Reclassification: ${asset.name} (${asset.assetNumber})`,
        sourceType: "manual", // Consistent with existing disposal JEs
        sourceId: asset._id as string,
        createdBy: ctx.user._id,
        lines: jeLines,
      });

      // Create expense record (per D-09)
      const expenseNumber = await getNextNumber(ctx, "EXP");
      const expenseId = await ctx.db.insert("expenses", {
        expenseNumber,
        submittedBy: args.submitterId,
        amount: nbv,
        accountId: targetExpenseAccountId,
        expenseDate: args.disposalDate,
        description: `Asset reclassified to expense: ${asset.name} (${asset.assetNumber})`,
        vendorName: "",
        paymentMethod: "company_paid" as const,
        status: "recorded" as const,
        lateSubmission: false,
        submittedAt: Date.now(),
        approvedBy: ctx.user._id,
        approvedAt: Date.now(),
        journalEntryId: jeId,
        sourceAssetId: asset._id,
        createdAt: Date.now(),
      });

      // Record audit trail
      await recordStatusChange(ctx, expenseId, undefined, "recorded", ctx.user._id, `Reclassified from asset: ${asset.assetNumber}`);

      // Update asset
      await ctx.db.patch(args.assetId, {

        disposalType: "reclassify_to_expense" as const,

      });

    }
