```typescript
// src/components/channelIntegration/AuditIssueTypeBadge.tsx
import { Tooltip, TooltipContent, TooltipTrigger } from "@/components/ui/tooltip";
import { Badge } from "@/components/ui/badge";

export type AuditIssueType = "unmapped_sku" | "stale_mapping" | "malformed_item" | "duplicate_transaction" | "orphan_item";

const TYPE_META: Record<AuditIssueType, { label: string; tooltip: string; severity: "warn" | "block" }> = {
  unmapped_sku: {
    label: "Unmapped SKU",
    tooltip: "Item has a product name but no linkedMenuProductId. Cannot deduct.",
    severity: "warn",
  },
  stale_mapping: {
    label: "Stale mapping",
    tooltip: "Item was mapped to a menu product that has since been renamed, delinked, or disabled.",
    severity: "warn",
  },
  malformed_item: {
    label: "Malformed item",
    tooltip: "Quantity ≤0, totalPrice <0, or required field missing.",
    severity: "block",
  },
  duplicate_transaction: {
    label: "Duplicate tx",
    tooltip: "Same (source, externalTransactionId, externalItemId) appears twice.",
    severity: "block",
  },
  orphan_item: {
    label: "Orphan item",
    tooltip: "externalRevenueItem.revenueId points to deleted or missing externalRevenue parent.",
    severity: "block",
  },
};

export function AuditIssueTypeBadge({ type }: { type: AuditIssueType }) {
  const meta = TYPE_META[type];
  return (
    <Tooltip>
      <TooltipTrigger asChild>
        <Badge variant={meta.severity === "block" ? "destructive" : "secondary"}>
          {meta.label}
        </Badge>
      </TooltipTrigger>
      <TooltipContent className="max-w-xs">{meta.tooltip}</TooltipContent>
    </Tooltip>
  );
}
```

```typescript
// src/hooks/convex/useChannelAudit.ts
import { useMutation, useQuery } from "convex/react";
import { api } from "@/convex/_generated/api";
import type { AuditIssueType } from "@/components/channelIntegration/AuditIssueTypeBadge";

export function useChannelAudit(token: string | undefined, filter?: { issueType?: AuditIssueType; source?: string; includeResolved?: boolean }) {
  const reports = useQuery(
    api.productInventory.channelAuditMutations.listAuditReports,
    token ? { token, limit: 5 } : "skip",
  );
  const issues = useQuery(
    api.productInventory.channelAuditMutations.listAuditIssues,
    token ? { token, ...filter } : "skip",
  );
  const resolveIssue = useMutation(api.productInventory.channelAuditMutations.resolveAuditIssue);
  const dismissIssue = useMutation(api.productInventory.channelAuditMutations.dismissAuditIssue);
  const triggerAudit = useMutation(api.productInventory.channelAuditMutations.triggerFullAudit);

  return { reports, issues, resolveIssue, dismissIssue, triggerAudit };
}
```

```typescript
// src/pages/ChannelAuditWorkbench.tsx
import { useState } from "react";
import { useChannelAudit } from "@/hooks/convex/useChannelAudit";
import { useAuth } from "@/hooks/useAuth";
import { AuditIssueTypeBadge, type AuditIssueType } from "@/components/channelIntegration/AuditIssueTypeBadge";
import { SourceBadge } from "@/components/channelIntegration/SourceBadge";
// ... shadcn UI imports
import { toast } from "sonner";
import { formatShortWIB } from "@/lib/dateUtils";

const ISSUE_TYPES: AuditIssueType[] = ["unmapped_sku", "stale_mapping", "malformed_item", "duplicate_transaction", "orphan_item"];

export default function ChannelAuditWorkbench() {
  const { token } = useAuth();
  const [activeTab, setActiveTab] = useState<AuditIssueType>("unmapped_sku");
  const [selected, setSelected] = useState<Set<string>>(new Set());
  const audit = useChannelAudit(token, { issueType: activeTab });
  const allIssues = useChannelAudit(token);  // for stat counts

  if (!token) return null;

  const countsByType = ISSUE_TYPES.reduce((acc, t) => {
    acc[t] = allIssues.issues?.filter(i => i.issueType === t && !i.resolvedAt).length ?? 0;
    return acc;
  }, {} as Record<AuditIssueType, number>);

  const lastReport = audit.reports?.[0];

  return (
    <div className="container mx-auto space-y-6 p-6">
      <PageHeader title="Channel Audit Workbench" description="Review and resolve data quality issues across all channel revenue items." />

      <div className="flex justify-between items-center">
        <Button
          onClick={async () => {
            try {
              await audit.triggerAudit({ token });
              toast.success("Audit scheduled — refresh in a few seconds for results.");
            } catch (err) {
              toast.error(err instanceof Error ? err.message : "Audit failed to schedule");
            }
          }}
        >Run full audit</Button>
        {lastReport && (
          <span className="text-sm text-muted-foreground">
            Last run {formatShortWIB(lastReport.startedAt)} · {lastReport.issuesFound} open
          </span>
        )}
      </div>

      {/* 5-stat grid */}
      <Card>
        <CardHeader><CardTitle>Summary</CardTitle></CardHeader>
        <CardContent>
          <div className="grid grid-cols-5 gap-4">
            {ISSUE_TYPES.map((t) => (
              <button
                key={t}
                onClick={() => setActiveTab(t)}
                className={cn("rounded-md border p-4 text-left", countsByType[t] > 0 ? "border-l-4 border-l-destructive" : "border-l-muted")}
              >
                <div className="text-sm font-semibold">{t}</div>
                <div className="text-2xl tabular-nums">{countsByType[t]}</div>
                <div className="text-xs text-muted-foreground">open</div>
              </button>
            ))}
          </div>
        </CardContent>
      </Card>

      <Tabs value={activeTab} onValueChange={(v) => { setActiveTab(v as AuditIssueType); setSelected(new Set()); }}>
        <TabsList>
          {ISSUE_TYPES.map((t) => (
            <TabsTrigger key={t} value={t}>
              <AuditIssueTypeBadge type={t} /> <span className="ml-2 tabular-nums">({countsByType[t]})</span>
            </TabsTrigger>
          ))}
        </TabsList>
        {ISSUE_TYPES.map((t) => (
          <TabsContent key={t} value={t}>
            <IssuesTable
              issues={audit.issues?.filter(i => i.issueType === t && !i.resolvedAt) ?? []}
              selected={selected}
              onSelect={setSelected}
              onResolve={async (id, resolution) => {
                await audit.resolveIssue({ token, issueId: id as any, resolution });
                toast.success("Issue resolved");
              }}
              onDismiss={async (id, reason) => {
                await audit.dismissIssue({ token, issueId: id as any, reason });
                toast.success("Issue dismissed");
              }}
            />
          </TabsContent>
        ))}
      </Tabs>
    </div>
  );
}
```
