```typescript
export function ChannelFlagRow({
  source,
  state,
  lastFlippedAt,
  lastFlippedBy,
  onFlip,

  state: FlagState;
  lastFlippedAt?: number;
```

```typescript
// convex/productInventory/channelFlags.ts
import { mutation, query } from "../_generated/server";
import { v } from "convex/values";
import { externalSource } from "../lib/externalSource";
import { requireRole } from "../lib/auth";

/** Read the 8-key flag map. Returns defaults (all false) if settings row absent. */
export const getChannelDeductionFlags = query({
  args: { token: v.string() },
  handler: async (ctx, args) => {
    await requireRole(ctx, args.token, ["admin"]);
    const settings = await ctx.db.query("productInventorySettings").first();
    const base = {
      bigseller: false, consignment: false, gobiz: false, grabfood: false,
      internal: false, k3mart: false, shopee: false, tiktok: false,
    };
    return { ...base, ...(settings?.channelDeductionEnabled ?? {}) };
  },
});

/** Toggle one flag. Persists across the full 8-key object shape. */
export const setChannelDeductionFlag = mutation({
  args: { token: v.string(), source: externalSource, enabled: v.boolean() },
  handler: async (ctx, args) => {
    const user = await requireRole(ctx, args.token, ["admin"]);

    const settings = await ctx.db.query("productInventorySettings").first();
    const current = settings?.channelDeductionEnabled ?? {
      bigseller: false, consignment: false, gobiz: false, grabfood: false,
      internal: false, k3mart: false, shopee: false, tiktok: false,
    };
    const next = { ...current, [args.source]: args.enabled };

    if (settings) {
      await ctx.db.patch(settings._id, {
        channelDeductionEnabled: next,
        updatedBy: user.name,
        updatedAt: Date.now(),
      });
    } else {
      // Settings row doesn't exist yet — create with sensible defaults + this flag flip.
      await ctx.db.insert("productInventorySettings", {
        globalLowStockThreshold: 5,
        autoAdvanceOnDrawdown: true,
        alertMode: "toast" as const,
        channelDeductionEnabled: next,
        updatedBy: user.name,
        updatedAt: Date.now(),
      });
    }

    return { source: args.source, enabled: args.enabled };
  },
});
```

```typescript
// src/pages/ProductInventorySettings.tsx
import { useMutation, useQuery } from "convex/react";
import { api } from "@/convex/_generated/api";
import { ChannelFlagRow } from "@/components/channelIntegration/ChannelFlagRow";
import { /* AlertDialog, PageHeader, Card, etc. */ } from "...";
import { useAuth } from "@/hooks/useAuth";
import { toast } from "sonner";

const DISABLED_REASONS = {
  bigseller: { reason: "na" as const, tooltip: "BigSeller writes parent revenue rows only. Per-item deduction is not supported." },
  grabfood: { reason: "blocked" as const, tooltip: "Requires GrabFood orders:read OAuth scope (pending)." },
  internal: { reason: "permanent-off" as const, tooltip: "Skipped to avoid double-deduction with reserveStockForOrderInternal. Override requires code review." },
};

export default function ProductInventorySettings() {
  const { token } = useAuth();
  const flags = useQuery(api.productInventory.channelFlags.getChannelDeductionFlags, token ? { token } : "skip");
  const setFlag = useMutation(api.productInventory.channelFlags.setChannelDeductionFlag);

  if (!flags) return <Skeleton className="h-48 w-full" />;

  const sources = ["shopee","tiktok","gobiz","k3mart","consignment","bigseller","grabfood","internal"] as const;

  return (
    <div className="container mx-auto space-y-6 p-6">
      <PageHeader title="Product Inventory Settings" description="Global thresholds, alert modes, and per-channel deduction flags." />

      <Card>
        <CardHeader>
          <CardTitle>Channel deduction</CardTitle>
          <CardDescription>Toggle inventory deduction per source. Flags default OFF.</CardDescription>
        </CardHeader>
        <CardContent>
          {sources.map((src) => {
            const disabled = DISABLED_REASONS[src as keyof typeof DISABLED_REASONS];
            const state = disabled
              ? { disabled: true as const, reason: disabled.reason, tooltip: disabled.tooltip }
              : { enabled: flags[src] };
            return (
              <ChannelFlagRow
                key={src}
                source={src}
                state={state}
                onFlip={async (next) => {
                  // TODO: show AlertDialog with "Type FLIP to confirm" if next=true
                  try {
                    await setFlag({ token, source: src, enabled: next });
                    toast.success(`${src} deduction turned ${next ? "on" : "off"}`);
                  } catch (err) {
                    toast.error(err instanceof Error ? err.message : "Flag flip failed");
                  }
                }}
              />
            );
          })}
          <div className="mt-4 text-sm text-muted-foreground tabular-nums">
            {Object.entries(flags).filter(([k, v]) => v && !DISABLED_REASONS[k as keyof typeof DISABLED_REASONS]).length}/5 eligible channels flipped ON
          </div>
        </CardContent>
      </Card>
    </div>
  );
}
```
