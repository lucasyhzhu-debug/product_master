import { describe, it, expect } from "vitest";
import { formatIndonesianDate } from "../dateUtils";

describe("formatIndonesianDate", () => {
  it("formats Date object as Indonesian full date", () => {
    // March 16, 2026 is a Monday (Senin)
    expect(formatIndonesianDate(new Date(2026, 2, 16))).toBe("Senin, 16 Maret 2026");
  });

  it("formats epoch milliseconds as Indonesian full date", () => {
    const epochMs = new Date(2026, 2, 16).getTime();
    expect(formatIndonesianDate(epochMs)).toBe("Senin, 16 Maret 2026");

  it("handles January 1st correctly", () => {
    // Jan 1, 2026 is a Thursday (Kamis)

  it("handles different day of week", () => {
    // March 15, 2026 is a Sunday (Minggu)

  });
});

/** Format date as Indonesian full date: "Senin, 16 Maret 2026" */
export function formatIndonesianDate(date: Date | number): string {
  const d = typeof date === "number" ? new Date(date) : date;
  return format(d, "EEEE, d MMMM yyyy", { locale: id });
}

Create the InvoiceFieldInput component -- a styled input that shows field source via background color.

**IMPORTANT: Use invoice-specific CSS custom property tokens for the color-coded backgrounds** instead of raw Tailwind color classes. This avoids dark mode breakage (CODE_STYLE.md prohibits raw Tailwind color classes for semantic backgrounds). The CONTEXT.md specifies `bg-blue-50` and `bg-amber-50` -- these are implemented as invoice-specific semantic tokens defined in index.css (see Task 2), NOT raw Tailwind classes.

```typescript
export type FieldSource = "auto" | "needs-input" | "edited";

// These classes reference CSS custom properties defined in src/index.css
// under the "Invoice Field Source Tokens" section.
// This is a deliberate invoice-specific semantic token set.
export const SOURCE_BG: Record<FieldSource, string> = {
  auto: "bg-[var(--invoice-field-auto)] border-[var(--invoice-field-auto-border)]",
  "needs-input": "bg-[var(--invoice-field-input)] border-[var(--invoice-field-input-border)]",
  edited: "bg-white border-gray-200",
};
```

Props:
- `label: string` -- field label (hidden in print)
- `value: string` -- current value
- `source: FieldSource` -- determines background color
- `onChange: (value: string) => void` -- change handler
- `placeholder?: string` -- optional placeholder
- `multiline?: boolean` -- if true, render Textarea instead of Input
- `readOnly?: boolean` -- if true, disable editing (for preview/print view embedding)

Component behavior:
- Render `Label` (with `print:hidden` class for labels) above the input
- Use shadcn `Input` or `Textarea` depending on `multiline` prop
- Apply `cn(SOURCE_BG[source], "print:border-none print:bg-transparent print:p-0 print:shadow-none print:h-auto")` for print-clean rendering
- The component does NOT manage source transitions -- the parent (InvoiceForm) handles transitioning "auto" -> "edited" when user types

**1. src/components/invoice/InvoicePrintView.tsx:**

Create a clean invoice print layout component with 9 sections per the design spec. This component is shared between on-screen preview (rendered in a bordered card) and the print route (rendered bare). It has NO form controls, NO colored field backgrounds -- just clean content.

**CRITICAL: Derive InvoicePrintData from the Convex Invoice query return type** instead of defining a standalone 25+ field interface. Read `src/hooks/convex/useInvoice.ts` to find the exact return type (the hook wraps `Doc<"invoices">`). Then define:

```typescript
import type { Doc } from "../../../convex/_generated/dataModel";

// Derive from Convex schema type to prevent drift
type InvoiceDoc = Doc<"invoices">;

export type InvoicePrintData = Pick<InvoiceDoc,
  | "invoiceNumber" | "generatedAt"
  | "sellerName" | "sellerAddress" | "sellerPhone" | "sellerEmail" | "sellerNpwp" | "sellerLogoUrl"
  | "bankName" | "bankAccountNumber" | "bankAccountName"
  | "buyerName" | "buyerCompany" | "buyerNpwp" | "buyerAddress" | "buyerPhone"
  | "poNumber" | "orderNumber" | "orderDate" | "dueDate"
  | "items" | "subtotal" | "discountAmount" | "discountLabel" | "deliveryFee" | "finalTotal"
  | "paymentStatus" | "paymentMethod" | "notes"
>;
```

This ensures InvoicePrintData stays in sync with the Convex schema automatically. If a field is renamed or removed in the schema, TypeScript will catch it here.

The component renders inside a `div` with BOTH `invoice-print-area` class (targeted by @media print CSS) AND serves as the wrapper for preview-from-print support.

9 sections (use semantic HTML with `aria-label` attributes on major sections, Tailwind for spacing/typography):

1. **Brand Bar**: `<div className="invoice-brand-bar h-1 bg-brand" />` -- thin accent bar at top. Use `h-1` (4px accent bar).

2. **Header** (flex row, `aria-label="Invoice header"`): Left side = logo image (if `sellerLogoUrl`, max-h-16, `loading="eager"` for print reliability) + seller name + address + phone + email + NPWP. Right side = large "INVOICE" title, invoice number (if finalized, else "DRAFT"), invoice date formatted via `formatIndonesianDate(generatedAt ?? Date.now())`.

3. **Bill To** (gray-50 background box, `aria-label="Buyer information"`): Buyer name, company (if set), NPWP (if set), billing address (if set), phone (if set).

4. **Order Details** (2-column grid, `aria-label="Order details"`): Order Number, Order Date (formatIndonesianDate), Due Date (formatIndonesianDate if set, else "-"), PO Number (if set, else "-").

5. **Items Table** (`aria-label="Invoice items"`): HTML `<table>` with headers: #, Product, Qty, Unit Price, Line Total. Each row shows item number, productName + variant (variant on second line in smaller text if set), qty, formatCurrency(unitPrice), formatCurrency(lineTotal). Use `border-collapse` and `border-b` for clean lines.

6. **Totals** (right-aligned summary, `aria-label="Invoice totals"`): Subtotal, Discount (show discountLabel like "Discount (10%)" if label set, else "Discount" -- only show row if discountAmount > 0), Delivery Fee (only show if > 0), Grand Total (bold, larger). Use `formatCurrency` for all amounts.

7. **Payment Info** (grid, `aria-label="Payment information"`): Bank Name, Account Number, Account Name (from seller bank details). Payment Status, Payment Method (if set).

8. **Signature Area**: Dashed border box (border-dashed border-gray-300), ~100px height, with text "Authorized Signature" and space below for stamp. This follows Indonesian business convention.

9. **Footer**: Centered text "Thank you for your business" + seller business name. Notes (if any) displayed above footer in italics.

Import `formatIndonesianDate` from `@/lib/dateUtils`, `formatCurrency` from `@/lib/utils`.
