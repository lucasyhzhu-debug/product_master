Replace with:
```typescript
export function useConvexOutletStockSummary(date: string) {
  const [data, setData] = useState<unknown>(undefined);
  const [isLoading, setIsLoading] = useState(true);
  const fetchAction = useAction(api.externalData.actions.fetchOutletStockSummary);

  const load = useCallback(async () => {
    setIsLoading(true);
    try {
      const result = await fetchAction({ date });
      setData(result);
    } catch (error) {
      console.error("Failed to fetch outlet stock summary:", error);
    } finally {
      setIsLoading(false);
    }
  }, [fetchAction, date]);

  useEffect(() => { load(); }, [load]);

  return { data, isLoading, refresh: load };
}
```

Add `useState`, `useCallback`, `useEffect` to the React imports. Add `useAction` to the `convex/react` import. Import `api` from the generated API if not already.
