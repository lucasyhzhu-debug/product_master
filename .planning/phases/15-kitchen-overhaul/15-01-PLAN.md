    ```typescript
    const DEFAULTS = {
      maxProductionTarget: 200,
      bigBallTarget: 150,
      midBallTarget: 50,
    };

    export const getConfig = query({
      args: {},
      handler: async (ctx) => {
        const config = await ctx.db.query("kitchenConfig").first();
        if (!config) {
          return {
            _id: null,
            maxProductionTarget: DEFAULTS.maxProductionTarget,
            bigBallTarget: DEFAULTS.bigBallTarget,
            midBallTarget: DEFAULTS.midBallTarget,
            updatedAt: null,
            updatedBy: null,
          };
        }
        return config;
      },
    });
    ```

    **Create convex/kitchenConfig/mutations.ts:**
    ```typescript
    import { mutation } from "../_generated/server";
    import { v } from "convex/values";
    import { requireRole } from "../lib/auth";

    export const updateConfig = mutation({
      args: {
        token: v.string(),
        maxProductionTarget: v.number(),
        bigBallTarget: v.number(),
        midBallTarget: v.number(),
      },
      handler: async (ctx, args) => {
        const user = await requireRole(ctx, args.token, ["manager", "admin"]);

        const existing = await ctx.db.query("kitchenConfig").first();
        const data = {
          maxProductionTarget: args.maxProductionTarget,
          bigBallTarget: args.bigBallTarget,
          midBallTarget: args.midBallTarget,
          updatedAt: Date.now(),
          updatedBy: user.name ?? "unknown",
        };

        if (existing) {
          await ctx.db.patch(existing._id, data);
          return existing._id;
        } else {
          return await ctx.db.insert("kitchenConfig", data);
        }
      },
    });
    ```

    ```typescript
    }
    ```

    **Add sendBackToOrderDesk mutation to convex/orders/mutations/kitchen.ts:**
    Add after the existing markOrderReady mutation:
    ```typescript
    export const sendBackToOrderDesk = mutation({
      args: {
        token: v.string(),
        orderId: v.id("orders"),
        reason: v.optional(v.string()),
      },
      handler: async (ctx, args) => {
        const user = await requireRole(ctx, args.token, ["kitchen", "manager", "admin"]);
        const order = await ctx.db.get(args.orderId);
        if (!order) throw new ConvexError("Order not found");
        if (order.status !== "BeingPrepared") {
          throw new ConvexError("Only orders being prepared can be sent back");
        }

        // Reset all package statuses to "empty"
        const items = await ctx.db
          .query("orderItems")
          .withIndex("by_order", (q) => q.eq("orderId", args.orderId))
          .collect();

        for (const item of items) {
          if (item.packageStatus && item.packageStatus !== "empty") {
            // Write unpack production log entry (matching togglePackOrderLineItem pattern)
            // to keep production counts accurate
            await ctx.db.insert("productionLog", {
              menuProductId: item.menuProductId,
              action: "unpack",
              quantity: item.quantity,
              timestamp: Date.now(),
              userId: user._id,
              orderId: args.orderId,
              orderItemId: item._id,
            });
            await ctx.db.patch(item._id, { packageStatus: "empty" });
          }
        }

        // Transition BeingPrepared -> PaymentReceived
        await ctx.db.patch(args.orderId, {
          status: "PaymentReceived",
          isKitchenVisible: computeIsKitchenVisible("PaymentReceived"),
        });

        // Log audit event
        await logAutoTransition(
          ctx,
          args.orderId,
          "BeingPrepared",
          "PaymentReceived",
          args.reason ?? "Sent back from kitchen",
          user._id
        );
      },
    });
    ```

    **IMPORTANT pitfall from research:** The `sendBackToOrderDesk` mutation MUST write `unpack` production log entries for each packed item. Without this, production counts will be skewed after a send-back. Check the existing `togglePackOrderLineItem` mutation for the exact productionLog entry pattern and match it (action field, required fields, etc.). The productionLog schema may use different field names -- read the table schema first.

    **IMPORTANT:** Do NOT reverse `confirmedAt` or touch sales aggregation. The `BeingPrepared -> PaymentReceived` transition does not cross the sales recognition boundary (per research finding #4).

    **Extend getKitchenStats in convex/orders/queries.ts:**
    The existing `getKitchenStats` query already returns `bigBallsNeeded`, `midBallsNeeded`, `ordersPending`, `ordersCompletedToday`. Extend it to also return:

    1. `minTarget`: Total balls needed from orders due today (confirmed/BeingPrepared with dueDate = today). This is the sum of BOM-derived ball counts for each order item where the order is due today.
    2. `minTargetOrderCount`: Count of orders due today that are in PaymentReceived or BeingPrepared status.
    3. `hasOverdueOrders`: Boolean -- true if any BeingPrepared orders have dueDate before today.

    To calculate min target:
    - Filter kitchen-visible orders (BeingPrepared) + PaymentReceived orders by `dueDate` matching today (use WIB timezone -- follow existing `wibNow` pattern if present, otherwise use `new Date()` with manual +7 offset).
    - For each order, sum the ball counts derived from BOM (menuProductComponents + componentTypes with category="production").
    - Return the total as `minTarget`.

    For `hasOverdueOrders`:
    - Check if any `BeingPrepared` orders have `dueDate < startOfToday()`.

    Add these fields to the return object of `getKitchenStats`. Do NOT break existing return shape -- only add new fields.
