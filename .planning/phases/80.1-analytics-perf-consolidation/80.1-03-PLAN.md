```tsx
import { useMemo } from "react";
import { ResponsiveHeatMap } from "@nivo/heatmap";
import { useDayHourHeatmap } from "@/hooks/convex/useAnalytics";
import { ChartFrame, ChartTooltip, formatCurrencyCompact } from "@/lib/chartPrimitives";

export function DayHourHeatmap() {
  const data = useDayHourHeatmap();

  const transformed = useMemo(() => {
    if (!data) return [];
    const { rowLabels, colLabels, grid } = data as any;
    return rowLabels.map((row: string, r: number) => ({
      id: row,
      data: colLabels.map((col: string, c: number) => ({
        x: col,
        y: grid[r]?.[c] ?? 0,
      })),
    }));
  }, [data]);

  if (data === undefined) {
    return (
      <ChartFrame title="Day × Hour heatmap" loading>
        {null}
      </ChartFrame>
    );
  }

  return (
    <ChartFrame title="Day × Hour heatmap" height={360}>
      <ResponsiveHeatMap
        data={transformed}
        margin={{ top: 20, right: 30, bottom: 48, left: 80 }}
        valueFormat={(v) => formatCurrencyCompact(Number(v))}
        axisTop={null}
        axisRight={null}
        axisBottom={{
          tickRotation: 0,
          legend: "Hour (WIB)",
          legendPosition: "middle",
          legendOffset: 36,
        }}
        axisLeft={{
          legend: "Day",
          legendPosition: "middle",
          legendOffset: -60,
        }}
        colors={{ type: "quantize", scheme: "purples", steps: 5 }}
        emptyColor="hsl(var(--muted))"
        labelTextColor={{ from: "color", modifiers: [["darker", 3]] }}
        tooltip={({ cell }) => (
          <ChartTooltip
            active
            label={String(cell.serieId)}
            payload={[
              {
                name: String(cell.data.x),
                value: Number(cell.value ?? 0),
              },
            ]}
          />
        )}
        theme={{
          text: { fill: "hsl(var(--foreground))" },
          axis: {
            ticks: { text: { fill: "hsl(var(--muted-foreground))" } },
            legend: { text: { fill: "hsl(var(--foreground))" } },
          },
        }}
      />
    </ChartFrame>
  );
}
```

```tsx
import { useMemo } from "react";
import { ResponsiveHeatMap } from "@nivo/heatmap";
import { useSkuChannelMatrix } from "@/hooks/convex/useAnalytics";
import { ChartFrame, ChartTooltip, truncateWithTooltip, formatCurrencyCompact } from "@/lib/chartPrimitives";

export function SkuChannelHeatmap({ topN = 8 }: { topN?: number }) {
  const data = useSkuChannelMatrix(topN);

  const { transformed, fullNameById } = useMemo(() => {
    const fullNameById = new Map<string, string>();
    if (!data) return { transformed: [] as any[], fullNameById };
    const rows = (data as any).rows ?? (data as any[]);
    const channels: string[] =
      (data as any).channels ??
      Array.from(
        new Set(
          rows.flatMap((r: any) =>
            r.cells ? r.cells.map((c: any) => c.channel) : Object.keys(r.channels ?? {}),
          ),
        ),
      );
    const transformed = rows.map((r: any) => {
      const display = truncateWithTooltip(r.productName, 22);
      fullNameById.set(display.display, display.full);
      const cells = channels.map((ch) => {
        const v = r.cells
          ? (r.cells.find((c: any) => c.channel === ch)?.value ?? 0)
          : (r.channels?.[ch] ?? 0);
        return { x: ch, y: v };
      });
      return { id: display.display, data: cells };
    });
    return { transformed, fullNameById };
  }, [data]);

  if (data === undefined) {
    return (
      <ChartFrame title="SKU × Channel heatmap" loading>
        {null}
      </ChartFrame>
    );
```
