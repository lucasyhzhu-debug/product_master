      args: { platformId: v.string() },
      handler: async (ctx, args) => {
        const cred = await ctx.db
          .query("platformCredentials")
          .withIndex("by_platform", (q) => q.eq("platformId", args.platformId))
          .first();
        return cred?.hmacSecret ?? null;
      },
    });

    **HMAC secret mutation in `convex/platformCredentials/mutations.ts`:**
    Add a new `saveHmacSecret` mutation (public, admin-only):
    ```typescript
    export const saveHmacSecret = mutation({
      args: {
        token: v.string(),
        platformId: v.string(),
        hmacSecret: v.string(),
      },
      handler: async (ctx, args) => {
        await requireRole(ctx, args.token, ["admin"]);
        const existing = await ctx.db
          .query("platformCredentials")
          .withIndex("by_platform", (q) => q.eq("platformId", args.platformId))
          .first();
        if (existing) {
          await ctx.db.patch(existing._id, {
            hmacSecret: args.hmacSecret,
            updatedBy: "admin",
            updatedAt: Date.now(),
          });
          return existing._id;
        }
        return await ctx.db.insert("platformCredentials", {
          platformId: args.platformId,
          hmacSecret: args.hmacSecret,
          updatedBy: "admin",
          updatedAt: Date.now(),
        });
      },
    });
    ```
