# Plan 69-01: Backend — Component Production Logging, Queries, and Schema Changes

<objective>
Extend the kitchen shift record system to support component-level production data (e.g., Outer-Marshmallow 500g, Filling-Pistachio 200g) alongside existing ball-level production (product quantities). Add a "component production" concept to kitchenShiftRecords, new queries to aggregate component production by date, and kitchenConfig changes for component visibility toggles.
</objective>

<must_haves>
- kitchenShiftRecords schema supports `componentProduced` array (component name, grams, optional ball type grouping)
- kitchenShiftRecords schema supports `componentWaste` array (component name, grams, reason)
- submitShiftRecord mutation accepts optional component production + waste data
- New query returns daily component production summary (aggregated across all shift records for a date)
- kitchenConfig supports a list of available kitchen components (name + optional ball type association)
- Component production data is backward-compatible (existing shift records without components continue to work)
- Component counts are NOT derived from BOM — they are independent manual inputs tracked in grams
</must_haves>

```typescript
import { ConvexError, v } from "convex/values";
import { requireRole } from "../lib/auth";

/**
 * Seed default kitchen components.
 * Called once to populate the initial list. Idempotent — skips if components already exist.
 */
export const seedDefaults = mutation({
  args: { token: v.string() },
  handler: async (ctx, args) => {
    await requireRole(ctx, args.token, ["admin"]);

    const existing = await ctx.db.query("kitchenComponents").first();
    if (existing) return { seeded: false, message: "Components already exist" };

    const now = Date.now();
    const defaults = [
      { name: "Outer-Marshmallow", code: "OUTER_MARSHMALLOW", ballTypeGroup: undefined, sortOrder: 1 },
      { name: "Filling-Pistachio", code: "FILLING_PISTACHIO", ballTypeGroup: undefined, sortOrder: 2 },
      { name: "Pistachio Spread", code: "PISTACHIO_SPREAD", ballTypeGroup: undefined, sortOrder: 3 },
      { name: "Salt", code: "SALT", ballTypeGroup: undefined, sortOrder: 4 },
      { name: "Marshmallow", code: "MARSHMALLOW", ballTypeGroup: undefined, sortOrder: 5 },
      { name: "Cacao Powder", code: "CACAO_POWDER", ballTypeGroup: undefined, sortOrder: 6 },
      { name: "Milk Powder", code: "MILK_POWDER", ballTypeGroup: undefined, sortOrder: 7 },
      { name: "Kunafa", code: "KUNAFA", ballTypeGroup: undefined, sortOrder: 8 },
      { name: "Pistachio Paste", code: "PISTACHIO_PASTE", ballTypeGroup: undefined, sortOrder: 9 },
      { name: "Butter", code: "BUTTER", ballTypeGroup: undefined, sortOrder: 10 },
    ];

    for (const comp of defaults) {
      await ctx.db.insert("kitchenComponents", {
        ...comp,
        unit: "g",
        isActive: true,
        createdAt: now,
        updatedAt: now,
      });
    }

    return { seeded: true, message: `Seeded ${defaults.length} kitchen components` };
  },
});

/**
 * Create a new kitchen component. Admin/manager only.
 */
export const create = mutation({
  args: {
    token: v.string(),
    name: v.string(),
    code: v.string(),
    ballTypeGroup: v.optional(v.string()),
    sortOrder: v.optional(v.number()),
  },
  handler: async (ctx, args) => {
    await requireRole(ctx, args.token, ["admin", "manager"]);

    // Check code uniqueness
    const existing = await ctx.db
      .query("kitchenComponents")
      .withIndex("by_code", (q) => q.eq("code", args.code))
      .first();
    if (existing) {
      throw new ConvexError(`Kitchen component with code "${args.code}" already exists`);
    }

    const now = Date.now();
    // Get max sortOrder for auto-assign
    const all = await ctx.db.query("kitchenComponents").collect();
    const maxSort = all.reduce((max, c) => Math.max(max, c.sortOrder), 0);

    return await ctx.db.insert("kitchenComponents", {
      name: args.name,
      code: args.code,
      ballTypeGroup: args.ballTypeGroup,
      unit: "g",
      isActive: true,
      sortOrder: args.sortOrder ?? maxSort + 1,
      createdAt: now,
      updatedAt: now,
    });
  },
});

/**
 * Update a kitchen component. Admin/manager only.
 */
export const update = mutation({
  args: {
    token: v.string(),
    id: v.id("kitchenComponents"),
    name: v.optional(v.string()),
    code: v.optional(v.string()),
    ballTypeGroup: v.optional(v.string()),
    isActive: v.optional(v.boolean()),
    sortOrder: v.optional(v.number()),
  },
  handler: async (ctx, args) => {
    await requireRole(ctx, args.token, ["admin", "manager"]);

    const existing = await ctx.db.get(args.id);
    if (!existing) throw new ConvexError("Kitchen component not found");

    // Check code uniqueness if changing
    if (args.code && args.code !== existing.code) {
      const dup = await ctx.db
        .query("kitchenComponents")
        .withIndex("by_code", (q) => q.eq("code", args.code))
        .first();
      if (dup) {
        throw new ConvexError(`Kitchen component with code "${args.code}" already exists`);
      }
    }

    const { token: _, id, ...updates } = args;
    const patchData: Record<string, unknown> = { updatedAt: Date.now() };
    if (updates.name !== undefined) patchData.name = updates.name;
    if (updates.code !== undefined) patchData.code = updates.code;
    if (updates.ballTypeGroup !== undefined) patchData.ballTypeGroup = updates.ballTypeGroup;
    if (updates.isActive !== undefined) patchData.isActive = updates.isActive;
    if (updates.sortOrder !== undefined) patchData.sortOrder = updates.sortOrder;

    await ctx.db.patch(args.id, patchData);
  },
});
```

```typescript
/**
 * List all kitchen components, optionally filtered to active only.
 * Sorted by sortOrder ascending.
 */
export const list = query({
  args: { activeOnly: v.optional(v.boolean()) },
  handler: async (ctx, args) => {
    let components;
    if (args.activeOnly) {
      components = await ctx.db
        .query("kitchenComponents")
        .withIndex("by_active", (q) => q.eq("isActive", true))
        .collect();
    } else {
      components = await ctx.db.query("kitchenComponents").collect();
    }
    return components.sort((a, b) => a.sortOrder - b.sortOrder);
  },
});

/**
 * Get a kitchen component by code.
 */
export const getByCode = query({
  args: { code: v.string() },
  handler: async (ctx, args) => {
    return await ctx.db
      .query("kitchenComponents")
      .withIndex("by_code", (q) => q.eq("code", args.code))
      .first();
  },
});
```

```typescript
/**
 * Get daily component production summary.
 * Aggregates componentProduced and componentWaste across all shift records for a date.
 * Returns per-component totals grouped by ball type, with per-person attribution.
 */
export const getDailyComponentSummary = query({
  args: { date: v.string() },
  handler: async (ctx, args) => {
    const records = await ctx.db
      .query("kitchenShiftRecords")
      .withIndex("by_date", (q) => q.eq("date", args.date))
      .collect();

    // Aggregate per component
    const componentMap = new Map<string, {
      code: string;
      name: string;
      totalProducedGrams: number;
      totalWasteGrams: number;
      perPerson: Array<{
        submittedBy: string;
        chefName?: string;
        producedGrams: number;
        wasteGrams: number;
      }>;
    }>();

    for (const record of records) {
      const personKey = record.chefName ?? record.submittedBy;

      // Process component produced
      if (record.componentProduced) {
        for (const item of record.componentProduced) {
          if (item.grams <= 0) continue;

          if (!componentMap.has(item.kitchenComponentCode)) {
            componentMap.set(item.kitchenComponentCode, {
              code: item.kitchenComponentCode,
              name: item.kitchenComponentName,
              totalProducedGrams: 0,
              totalWasteGrams: 0,
              perPerson: [],
            });
          }
          const entry = componentMap.get(item.kitchenComponentCode)!;
          entry.totalProducedGrams += item.grams;

          // Per-person attribution
          const existing = entry.perPerson.find(p => (p.chefName ?? p.submittedBy) === personKey);
          if (existing) {
            existing.producedGrams += item.grams;
          } else {
            entry.perPerson.push({
              submittedBy: record.submittedBy,
              chefName: record.chefName,
              producedGrams: item.grams,
              wasteGrams: 0,
            });
          }
        }
      }

      // Process component waste
      if (record.componentWaste) {
        for (const item of record.componentWaste) {
          if (item.grams <= 0) continue;

          if (!componentMap.has(item.kitchenComponentCode)) {
            componentMap.set(item.kitchenComponentCode, {
              code: item.kitchenComponentCode,
              name: item.kitchenComponentName,
              totalProducedGrams: 0,
              totalWasteGrams: 0,
              perPerson: [],
            });
          }
          const entry = componentMap.get(item.kitchenComponentCode)!;
          entry.totalWasteGrams += item.grams;

          // Per-person waste attribution
          const existing = entry.perPerson.find(p => (p.chefName ?? p.submittedBy) === personKey);
          if (existing) {
            existing.wasteGrams += item.grams;
          } else {
            entry.perPerson.push({
              submittedBy: record.submittedBy,
              chefName: record.chefName,
              producedGrams: 0,
              wasteGrams: item.grams,
            });
          }
        }
      }
    }

    return Array.from(componentMap.values());
  },
});
```
</action>
