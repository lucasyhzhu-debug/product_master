```typescript
/**
 * Phase 74.5.2 Plan 03 — Backfill integration tests.
 *
 * Covers R8 (timestamp preservation), SC3 (idempotency), Pitfall 3 (null-menuProduct
 * skip preserves inventoryDeductedAt=undefined), and admin-only gate on
 * runChannelBackfill.
 */
import { convexTest } from "convex-test";
import { expect, test, describe } from "vitest";
import schema from "../../schema";
import { api, internal } from "../../_generated/api";
import type { Id } from "../../_generated/dataModel";

// Copy the exact glob pattern from channelSale.test.ts after Plan 01 fix.
// If Plan 01 used Fix 1 (./*.{ts,tsx}), replicate:
const modules = import.meta.glob("../../**/*.{ts,tsx}");

describe("backfillChannelDeductions", () => {
  test("idempotent: re-run after completion produces zero double-deductions (SC3)", async () => {
    const t = convexTest(schema, modules);
    // Seed prerequisites: outlet, menuProduct, storageLocation, routing rule.
    const { menuProductId, revenueId, itemId } = await t.run(async (ctx) => {
      const locationId = await ctx.db.insert("storageLocations", {
        name: "Test Depot",
        code: "test_depot",
        locationType: "depot",
        isActive: true,
        createdAt: Date.now(),
      });
      const menuProductId = await ctx.db.insert("menuProducts", {
        name: "Test Product",
        sku: "TEST-001",
        recipeId: null,
        isActive: true,
        createdAt: Date.now(),
      });
      // Seed stock so the deduct succeeds.
      await ctx.db.insert("productInventoryTransactions", {
        menuProductId,
        locationId,
        transactionType: "add",
        quantity: 100,
        previousQuantity: 0,
        newQuantity: 100,
        performedBy: "system:test_seed",
        createdAt: Date.now() - 1000,
      });
      // Routing rule: shopee → locationId, tier 2 (source-default).
      await ctx.db.insert("channelRouting", {
        source: "shopee",
        tier: 2,
        outletId: undefined,
        menuProductId: undefined,
        locationId,
        createdAt: Date.now(),
        updatedAt: Date.now(),
      });
      const revenueId = await ctx.db.insert("externalRevenue", {
        source: "shopee",
        externalTransactionId: "shopee-tx-1",
        transactionDate: 1700000000000,
        totalAmount: 100000,
        createdAt: Date.now(),
      });
      const itemId = await ctx.db.insert("externalRevenueItems", {
        revenueId,
        source: "shopee",
        externalItemId: "shopee-item-1",
        productName: "Test Product",
        quantity: 5,
        unitPrice: 20000,
        totalPrice: 100000,
        linkedMenuProductId: menuProductId,
        inventoryDeductedAt: undefined,
        createdAt: Date.now(),
      });
      return { menuProductId, revenueId, itemId };
    });

    // First run: should deduct.
    const firstResult = await t.run(async (ctx) => {
      return await ctx.runMutation(
        internal.productInventory.backfill.backfillOnePage,
        { source: "shopee" },
      );
    });
    expect(firstResult.itemsProcessed).toBe(1);
    expect(firstResult.deducted).toBe(1);
    expect(firstResult.skipped).toBe(0);

    // Verify one channel_sale ledger row exists with createdAt === transactionDate.
    await t.run(async (ctx) => {
      const txRows = await ctx.db
        .query("productInventoryTransactions")
        .filter((q) => q.eq(q.field("transactionType"), "channel_sale"))
        .collect();
      expect(txRows.length).toBe(1);
      expect(txRows[0].source).toBe("shopee");
      expect(txRows[0].quantity).toBe(-5);
      expect(txRows[0].createdAt).toBe(1700000000000);
      const item = await ctx.db.get(itemId);
      expect(item?.inventoryDeductedAt).toBeDefined();
      expect(item?.inventoryDeductedAt).toBeTypeOf("number");
    });

    // Second run: should be no-op.
    const secondResult = await t.run(async (ctx) => {
      return await ctx.runMutation(
        internal.productInventory.backfill.backfillOnePage,
        { source: "shopee" },
      );
    });
    expect(secondResult.itemsProcessed).toBe(0);
    expect(secondResult.deducted).toBe(0);

    // Still exactly one ledger row.
    await t.run(async (ctx) => {
      const txRows = await ctx.db
        .query("productInventoryTransactions")
        .filter((q) => q.eq(q.field("transactionType"), "channel_sale"))
        .collect();
      expect(txRows.length).toBe(1);
    });
  });

  test("preserves revenue.transactionDate as ledger createdAt (R8)", async () => {
    const t = convexTest(schema, modules);
    const FIXED_DATE = 1700000000000;
    await t.run(async (ctx) => {
      const locationId = await ctx.db.insert("storageLocations", {
        name: "Depot", code: "depot", locationType: "depot",
        isActive: true, createdAt: Date.now(),
      });
      const menuProductId = await ctx.db.insert("menuProducts", {
        name: "P", sku: "P-1", recipeId: null, isActive: true, createdAt: Date.now(),
      });
      await ctx.db.insert("productInventoryTransactions", {
        menuProductId, locationId, transactionType: "add",
        quantity: 100, previousQuantity: 0, newQuantity: 100,
        performedBy: "seed", createdAt: Date.now() - 1000,
      });
      await ctx.db.insert("channelRouting", {
        source: "tiktok", tier: 2, locationId,
        createdAt: Date.now(), updatedAt: Date.now(),
      });
      const revenueId = await ctx.db.insert("externalRevenue", {
        source: "tiktok", externalTransactionId: "tt-1",
        transactionDate: FIXED_DATE, totalAmount: 50000, createdAt: Date.now(),
      });
      await ctx.db.insert("externalRevenueItems", {
        revenueId, source: "tiktok", externalItemId: "tt-item-1",
        productName: "P", quantity: 2, unitPrice: 25000, totalPrice: 50000,
        linkedMenuProductId: menuProductId,
        inventoryDeductedAt: undefined, createdAt: Date.now(),
      });
    });

    await t.run(async (ctx) => {
      return await ctx.runMutation(
        internal.productInventory.backfill.backfillOnePage,
        { source: "tiktok" },
      );
    });

    await t.run(async (ctx) => {
      const tx = (await ctx.db
        .query("productInventoryTransactions")
        .filter((q) => q.eq(q.field("transactionType"), "channel_sale"))
        .collect())[0];
      expect(tx.createdAt).toBe(FIXED_DATE);
      expect(tx.createdAt).not.toBe(Date.now()); // paranoia: NOT current time
    });
  });

  test("null linkedMenuProductId: skip but do NOT patch inventoryDeductedAt (Pitfall 3)", async () => {
    const t = convexTest(schema, modules);
    const { itemId } = await t.run(async (ctx) => {
      const revenueId = await ctx.db.insert("externalRevenue", {
        source: "bigseller", externalTransactionId: "bs-1",
        transactionDate: Date.now(), totalAmount: 30000, createdAt: Date.now(),
      });
      const itemId = await ctx.db.insert("externalRevenueItems", {
        revenueId, source: "bigseller", externalItemId: "bs-item-1",
        productName: "Unmapped", quantity: 1, unitPrice: 30000, totalPrice: 30000,
        linkedMenuProductId: undefined, // KEY: unmapped
        inventoryDeductedAt: undefined, createdAt: Date.now(),
      });
      return { itemId };
    });

    const result = await t.run(async (ctx) => {
      return await ctx.runMutation(
        internal.productInventory.backfill.backfillOnePage,
        { source: "bigseller" },
      );
    });
    expect(result.itemsProcessed).toBe(1);
    expect(result.deducted).toBe(0);
    expect(result.skipped).toBe(1);

    await t.run(async (ctx) => {
      const txRows = await ctx.db
        .query("productInventoryTransactions")
        .filter((q) => q.eq(q.field("transactionType"), "channel_sale"))
        .collect();
      expect(txRows.length).toBe(0);
      const item = await ctx.db.get(itemId);
      // CRITICAL: inventoryDeductedAt must still be undefined so admin can re-attempt after mapping.
      expect(item?.inventoryDeductedAt).toBeUndefined();
    });
  });

  test("runChannelBackfill rejects non-admin token", async () => {
    const t = convexTest(schema, modules);
    const { kitchenToken, adminToken } = await t.run(async (ctx) => {
      const kitchenUserId = await ctx.db.insert("users", {
        name: "Kitchen", role: "kitchen", pin: "1234", isActive: true, createdAt: Date.now(),
      });
      const kitchenSessionId = await ctx.db.insert("sessions", {
        userId: kitchenUserId, token: "kitchen-token", expiresAt: Date.now() + 86400000, createdAt: Date.now(),
      });
      const adminUserId = await ctx.db.insert("users", {
        name: "Admin", role: "admin", pin: "9999", isActive: true, createdAt: Date.now(),
      });
      const adminSessionId = await ctx.db.insert("sessions", {
        userId: adminUserId, token: "admin-token", expiresAt: Date.now() + 86400000, createdAt: Date.now(),
      });
      return { kitchenToken: "kitchen-token", adminToken: "admin-token" };
    });

    await expect(
      t.mutation(api.productInventory.backfill.runChannelBackfill, {
        source: "shopee", token: kitchenToken,
      }),
    ).rejects.toThrow();

    const result = await t.mutation(
      api.productInventory.backfill.runChannelBackfill,
      { source: "shopee", token: adminToken },
    );
    expect(result).toEqual({ scheduled: true });
  });
});
```
