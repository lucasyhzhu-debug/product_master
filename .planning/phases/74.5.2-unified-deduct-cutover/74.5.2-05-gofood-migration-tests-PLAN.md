```typescript
/**
 * Phase 74.5.2 Plan 05 — gofood_sale → channel_sale migration tests.
 *
 * Covers:
 *   - D74.5.2-L6 forward-only semantics
 *   - Pitfall 1 landmine (source: "gobiz" NOT "gofood")
 *   - gofoodOrderRef preservation + externalRef derivation
 *   - Self-healing on re-run
 *   - Admin-only gate on public runner
 */
import { convexTest } from "convex-test";
import { expect, test, describe } from "vitest";
import schema from "../../schema";
import { api, internal } from "../../_generated/api";

const modules = import.meta.glob("../../**/*.{ts,tsx}");

// Helper — seed a minimal gofood_sale row. Adjust required fields to match actual schema.
async function seedGofoodSale(
  ctx: any,
  opts: {
    menuProductId?: any;
    locationId?: any;
    gofoodOrderRef?: string;
    quantity?: number;
  } = {},
): Promise<any> {
  const locationId = opts.locationId ?? await ctx.db.insert("storageLocations", {
    name: "Test Depot", code: "test", locationType: "depot",
    isActive: true, createdAt: Date.now(),
  });
  const menuProductId = opts.menuProductId ?? await ctx.db.insert("menuProducts", {
    name: "Test", sku: "T-1", recipeId: null, isActive: true, createdAt: Date.now(),
  });
  return await ctx.db.insert("productInventoryTransactions", {
    menuProductId,
    locationId,
    transactionType: "gofood_sale",
    quantity: -(opts.quantity ?? 1),
    previousQuantity: 10,
    newQuantity: 10 - (opts.quantity ?? 1),
    gofoodOrderRef: opts.gofoodOrderRef,
    performedBy: "system:gobiz_sync",
    createdAt: Date.now(),
  });
}

describe("migrateGofoodSaleToChannelSale", () => {
  test("chunked migration rewrites all gofood_sale rows to channel_sale+gobiz", async () => {
    const t = convexTest(schema, modules);
    await t.run(async (ctx) => {
      for (let i = 0; i < 10; i++) {
        await seedGofoodSale(ctx, { gofoodOrderRef: `go-${i}` });
      }
    });

    // Run the migration — invoke the action via its handler or via the internal API.
    await t.run(async (ctx) => {
      // Loop one page at a time (simpler than scheduling — test harness).
      let cursor: string | null = null;
      while (true) {
        const result = await ctx.runMutation(
          internal.migrations.gofoodSaleToChannelSale.migrateOnePage,
          { paginationOpts: { numItems: 500, cursor } },
        );
        if (result.isDone) break;
        cursor = result.continueCursor;
      }
    });

    // Verify zero gofood_sale rows remain.
    await t.run(async (ctx) => {
      const remaining = await ctx.db
        .query("productInventoryTransactions")
        .withIndex("by_type", (q) => q.eq("transactionType", "gofood_sale"))
        .collect();
      expect(remaining.length).toBe(0);

      const migrated = await ctx.db
        .query("productInventoryTransactions")
        .withIndex("by_type", (q) => q.eq("transactionType", "channel_sale"))
        .collect();
      expect(migrated.length).toBe(10);
      for (const row of migrated) {
        expect(row.source).toBe("gobiz");
      }
    });
  });

  test("writes source: gobiz (NOT gofood — landmine Pitfall 1)", async () => {
    const t = convexTest(schema, modules);
    await t.run(async (ctx) => {
      await seedGofoodSale(ctx, { gofoodOrderRef: "go-lm-1" });
    });

    await t.run(async (ctx) => {
      await ctx.runMutation(
        internal.migrations.gofoodSaleToChannelSale.migrateOnePage,
        { paginationOpts: { numItems: 500, cursor: null } },
      );
    });

    await t.run(async (ctx) => {
      const rows = await ctx.db
        .query("productInventoryTransactions")
        .withIndex("by_type", (q) => q.eq("transactionType", "channel_sale"))
        .collect();
      expect(rows.length).toBe(1);
      expect(rows[0].source).toBe("gobiz");
      // Extra: ensure no row accidentally wrote source="gofood"
      const gofoodSourced = rows.filter((r) => (r.source as any) === "gofood");
      expect(gofoodSourced.length).toBe(0);
    });
  });

  test("preserves gofoodOrderRef and derives externalRef with fallback", async () => {
    const t = convexTest(schema, modules);
    const ids = await t.run(async (ctx) => {
      const id1 = await seedGofoodSale(ctx, { gofoodOrderRef: "go-ref-A" });
      const id2 = await seedGofoodSale(ctx, { gofoodOrderRef: "go-ref-B" });
      const id3 = await seedGofoodSale(ctx, { gofoodOrderRef: undefined }); // fallback case
      return { id1, id2, id3 };
    });

    await t.run(async (ctx) => {
      await ctx.runMutation(
        internal.migrations.gofoodSaleToChannelSale.migrateOnePage,
        { paginationOpts: { numItems: 500, cursor: null } },
      );
    });

    await t.run(async (ctx) => {
      const r1 = await ctx.db.get(ids.id1);
      expect(r1?.externalRef).toBe("go-ref-A");
      expect(r1?.gofoodOrderRef).toBe("go-ref-A"); // preserved, NOT cleared

      const r2 = await ctx.db.get(ids.id2);
      expect(r2?.externalRef).toBe("go-ref-B");
      expect(r2?.gofoodOrderRef).toBe("go-ref-B");

      const r3 = await ctx.db.get(ids.id3);
      expect(r3?.externalRef).toMatch(/^legacy-/);
      expect(r3?.gofoodOrderRef).toBeUndefined();
    });
  });

  test("self-healing: re-run on partial completion is a no-op", async () => {
    const t = convexTest(schema, modules);
    await t.run(async (ctx) => {
      for (let i = 0; i < 5; i++) {
        await seedGofoodSale(ctx, { gofoodOrderRef: `go-sh-${i}` });
      }
    });

    // First run — migrates all 5.
    const firstRun = await t.run(async (ctx) => {
      return await ctx.runMutation(
        internal.migrations.gofoodSaleToChannelSale.migrateOnePage,
        { paginationOpts: { numItems: 500, cursor: null } },
      );
    });
    expect(firstRun.migrated).toBe(5);

    // Second run — should migrate zero.
    const secondRun = await t.run(async (ctx) => {
      return await ctx.runMutation(
        internal.migrations.gofoodSaleToChannelSale.migrateOnePage,
        { paginationOpts: { numItems: 500, cursor: null } },
      );
    });
    expect(secondRun.migrated).toBe(0);

    // Row count integrity.
    await t.run(async (ctx) => {
      const channelSaleRows = await ctx.db
        .query("productInventoryTransactions")
        .withIndex("by_type", (q) => q.eq("transactionType", "channel_sale"))
        .collect();
      expect(channelSaleRows.length).toBe(5);
    });
  });

  test("runGofoodSaleToChannelSaleMigration rejects non-admin token", async () => {
    const t = convexTest(schema, modules);
    const { adminToken, kitchenToken } = await t.run(async (ctx) => {
      const adminId = await ctx.db.insert("users", {
        name: "Admin", role: "admin", pin: "9999", isActive: true, createdAt: Date.now(),
      });
      await ctx.db.insert("sessions", {
        userId: adminId, token: "admin-token", expiresAt: Date.now() + 86400000, createdAt: Date.now(),
      });
      const kitchenId = await ctx.db.insert("users", {
        name: "Kitchen", role: "kitchen", pin: "1234", isActive: true, createdAt: Date.now(),
      });
      await ctx.db.insert("sessions", {
        userId: kitchenId, token: "kitchen-token", expiresAt: Date.now() + 86400000, createdAt: Date.now(),
      });
      return { adminToken: "admin-token", kitchenToken: "kitchen-token" };
    });

    await expect(
      t.mutation(
        api.migrations.gofoodSaleToChannelSale.runGofoodSaleToChannelSaleMigration,
        { token: kitchenToken },
      ),
    ).rejects.toThrow();

    const result = await t.mutation(
      api.migrations.gofoodSaleToChannelSale.runGofoodSaleToChannelSaleMigration,
      { token: adminToken },
    );
    expect(result).toEqual({ scheduled: true });
  });
});
```
