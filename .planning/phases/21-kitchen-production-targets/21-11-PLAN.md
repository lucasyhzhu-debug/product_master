Update the production summary to show delta per product:
```tsx
<ul className="space-y-2">
  {produced.map((item) => {
    const target = targetMap.get(item.menuProductId) ?? 0;
    // Waste for this product counts toward "total made"
    const wasteForProduct = waste
      .filter(w => w.menuProductId === item.menuProductId)
      .reduce((sum, w) => sum + w.quantity, 0);
    const totalMade = item.quantity + wasteForProduct;
    const delta = totalMade - target;
    const deltaColor = delta >= 0 ? "text-emerald-600" : "text-amber-600";
    const deltaSign = delta >= 0 ? "+" : "";

    return (
      <li key={item.menuProductId} className="rounded-lg border border-border bg-card p-3">
        <div className="flex items-center justify-between">
          <span className="text-sm font-medium">{item.menuProductName}</span>
          <div className="text-right">
            <span className="text-sm font-semibold tabular-nums">{item.quantity} produced</span>
            {wasteForProduct > 0 && (
              <span className="text-xs text-muted-foreground ml-1">+ {wasteForProduct} waste</span>
            )}
          </div>
        </div>
        {target > 0 && (
          <div className="flex items-center justify-between mt-1">
            <span className="text-xs text-muted-foreground">Target: {target}</span>
            <span className={`text-xs font-medium tabular-nums ${deltaColor}`}>
              {deltaSign}{delta} ({totalMade}/{target})
            </span>
          </div>
        )}
      </li>
    );
  })}
</ul>
```

const container = {
  hidden: {},
  show: {
    transition: {
      staggerChildren: 0.12,
      delayChildren: 0.2,
    },
  },
};

const item = {
  hidden: { opacity: 0, x: -10 },
  show: { opacity: 1, x: 0, transition: { duration: 0.3 } },
};

return (
  <Card className="border-emerald-200 bg-emerald-50/50 dark:bg-emerald-950/20 dark:border-emerald-800">
    <CardContent className="pt-6 pb-4 space-y-4">
      <div className="flex items-center gap-3">

        <h3 className="text-lg font-semibold text-foreground">Shift Recorded</h3>

      <motion.div
        variants={container}
        initial="hidden"
        animate="show"
        className="space-y-2"

          Production
        </p>
        {produced.map((p) => (
          <motion.div
            key={p.menuProductId}
            variants={item}

          >
            <div className="flex items-center gap-2">
              <CheckCircle2 className="h-4 w-4 text-emerald-500 shrink-0" />
              <span className="text-sm font-medium">{p.menuProductName}</span>
            </div>
            <span className="text-sm font-semibold tabular-nums">{p.quantity} units</span>

        ))}
      </motion.div>

      {waste.length > 0 && (

            Waste
          </p>
          {waste.map((w, idx) => (

              key={`${w.menuProductId}-${w.reason}-${idx}`}
              variants={item}

              <div className="flex items-center gap-2">
                <span className="text-sm">{w.menuProductName}</span>

                  ({REASON_LABELS[w.reason] ?? w.reason})
                </span>
              </div>
              <span className="text-sm font-semibold tabular-nums text-destructive">
                -{w.quantity}
              </span>

      <p className="text-xs text-muted-foreground">
        Inventory has been updated at the Kitchen location.

        Done
      </Button>

  </Card>
