const visibleItems = packagingItems.filter(item => {
  if (!enabledComponents || !productBallTypes) return true;
  const ballTypes = productBallTypes[item.menuProductId] ?? [];
  // Show if at least one of its ball types is enabled
  return ballTypes.some(bt => enabledComponents.includes(bt));

const flaggedItems = packagingItems.filter(item => {

**KitchenViewV2 wiring:**

1. Query `menuProductComponents` and `componentTypes` (production category) to build `productBallTypes` map:
```typescript
const menuProductComps = useQuery(api.menuProductComponents.queries.listAll);
const componentTypes = useQuery(api.componentTypes.queries.list);

const productBallTypes = useMemo(() => {
  if (!menuProductComps || !componentTypes) return undefined;
  const prodTypes = componentTypes.filter(ct => ct.category === "production");
  const codeMap = new Map(prodTypes.map(ct => [String(ct._id), ct.code]));

  const result: Record<string, string[]> = {};
  for (const comp of menuProductComps) {
    if (!comp.componentTypeId) continue;
    const code = codeMap.get(String(comp.componentTypeId));
    if (!code) continue;
    const key = String(comp.menuProductId);
    if (!result[key]) result[key] = [];
    if (!result[key].includes(code)) result[key].push(code);
  }
  return result;
}, [menuProductComps, componentTypes]);
```
